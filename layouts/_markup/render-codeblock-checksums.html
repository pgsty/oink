{{- /* ```checksums {base="https://…/download/v1.7.0/" algo="sha256" group="auto"} — sha*sum lines as a data fence: the native form of release-assets (Release pack, v5 §3.10). Same parser and renderers as the shortcode; `base` is required unless the page has release front matter. A malformed fence warns and renders nothing. Guessing at an algorithm or a base would publish rows pointing at the wrong files, and stopping the build would take the whole site down over one fence. */ -}} {{- $page := .PageInner | default .Page -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- $policy := partial "content/attributes.html" (dict "attributes" .Attributes "allowed" (slice "base" "algo" "group") "position" .Position "kind" "checksums") -}} {{- $ok := true -}} {{- $algo := index $policy.consumed "algo" | default "" -}} {{- if and $algo (not (in (slice "md5" "sha1" "sha256" "sha512") $algo)) -}} {{- warnf "checksums: algo must be md5, sha1, sha256, or sha512 at %s; skipping the block" .Position -}} {{- $ok = false -}} {{- end -}} {{- $grouped := false -}} {{- with index $policy.consumed "group" -}} {{- if ne . "auto" -}} {{- warnf "checksums: group must be auto at %s; ignoring it" $.Position -}} {{- else -}} {{- $grouped = true -}} {{- end -}} {{- end -}} {{- $inner := printf "%s" .Inner -}} {{- if and $ok (not (strings.ContainsNonSpace $inner)) -}} {{- warnf "checksums: the fence requires checksum lines at %s; skipping the block" .Position -}} {{- $ok = false -}} {{- end -}} {{- $base := "" -}} {{- $hasRelease := isset $page.Params "release" -}} {{- if $ok -}} {{- with index $policy.consumed "base" -}} {{- $baseInput := strings.TrimSpace . -}} {{- if $hasRelease -}} {{- warnf "checksums: base is only valid without release front matter at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else if not $baseInput -}} {{- warnf "checksums: base must not be empty at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else if findRE `[\x00-\x20\x7f]` $baseInput -}} {{- warnf "checksums: base must not contain whitespace or control characters at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else if hasPrefix $baseInput "//" -}} {{- warnf "checksums: base must not use a protocol-relative URL at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else -}} {{- $parsedResult := try (urls.Parse $baseInput) -}} {{- if $parsedResult.Err -}} {{- warnf "checksums: invalid base URL at %s: %s; skipping the block" $.Position $parsedResult.Err -}} {{- $ok = false -}} {{- else -}} {{- $parsed := $parsedResult.Value -}} {{- if and $parsed.Scheme (not (in (slice "http" "https") (lower $parsed.Scheme))) -}} {{- warnf "checksums: base URL must use http or https at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else if and $parsed.Scheme (not $parsed.Host) -}} {{- warnf "checksums: base URL requires a host at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else if or $parsed.RawQuery $parsed.Fragment -}} {{- warnf "checksums: base URL must not contain a query or fragment at %s; skipping the block" $.Position -}} {{- $ok = false -}} {{- else -}} {{- $base = $baseInput -}} {{- if not $parsed.Scheme }}{{ $base = strings.TrimPrefix "/" $baseInput | relURL }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if and $ok (not $base) -}} {{- if $hasRelease -}} {{- $meta := partial "release/meta.html" (dict "page" $page "position" .Position) -}} {{- if not $meta.valid -}} {{- /* meta.html has already warned about the front matter itself. */ -}} {{- $ok = false -}} {{- else -}} {{- $base = printf "%s/releases/download/%s" $meta.repoURL (urls.PathEscape $meta.tag) -}} {{- end -}} {{- else -}} {{- warnf "checksums: base is required unless the page has release front matter at %s; skipping the block" .Position -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $entries := partial "release/assets-parse.html" (dict "content" $inner "position" .Position "base" $base "algo" $algo) -}} {{- if in (slice "markdown" "rss") $format -}} {{- partial "release/assets-markdown.html" (dict "entries" $entries "grouped" $grouped) | safeHTML -}} {{- else -}} {{- $interactive := eq $format "html" -}} {{- if $interactive }}{{ $page.Store.Set "hasAssetList" true }}{{ end -}} {{- partial "release/assets-table.html" (dict "entries" $entries "grouped" $grouped "interactive" $interactive "class" $policy.class "generic" $policy.generic ) -}} {{- end -}} {{- end -}}
