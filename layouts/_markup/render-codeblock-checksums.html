}}{{ errorf "checksums: the fence requires checksum lines at %s" .Position }}{{ end -}} {{- $base := "" -}} {{- $hasRelease := isset $page.Params "release" -}} {{- with index $policy.consumed "base" -}} {{- $baseInput := strings.TrimSpace . -}} {{- if $hasRelease }}{{ errorf "checksums: base is only valid without release front matter at %s" $.Position }}{{ end -}} {{- if not $baseInput }}{{ "checksums: base URL requires a host at %s" $.Position }}{{ end -}} {{- if or $parsed.RawQuery $parsed.Fragment }}{{ errorf "checksums: base URL must not contain a query or fragment at %s" $.Position := partial "release/meta.html" (dict "page" $page "position" .Position) -}} {{- $base = printf "%s/releases/download/%s" $meta.repoURL (urls.PathEscape $meta.tag) -}} {{- else -}} {{- errorf -}} {{- $interactive := eq $format "html" -}} {{- if $interactive }}{{ $page.Store.Set "hasAssetList" true }}{{ end -}} {{- partial "release/assets-table.html" (dict "entries" $entries "grouped"
