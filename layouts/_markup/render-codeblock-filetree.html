{{- /* ```filetree {title="…"} — annotated directory tree as a data fence. The body is parsed by content/filetree-parse.html; rendering is CSS + native
. Interactive HTML gets collapsible directories, a build-time aligned comment column and — when there are comments — a draggable split (`hasFileTree` loads assets/js/filetree.js); print gets the same tree fully expanded and static; Markdown re-emits the fence; RSS gets the source as 
. */ -}}
{{- $page := .PageInner | default .Page -}}
{{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}}
{{- $policy := partial "content/attributes.html" (dict
  "attributes" .Attributes "allowed" (slice "title" "tab" "group" "value")
  "position" .Position "kind" "filetree"
) -}}
{{- $title := strings.TrimSpace (index $policy.consumed "title" | default "") -}}
{{- if and (isset $policy.consumed "title") (not $title) }}{{ warnf "filetree: title must not be empty at %s; dropping it" .Position }}{{ end -}}
{{- $tab := strings.TrimSpace (index $policy.consumed "tab" | default "") -}}
{{- if and (isset $policy.consumed "tab") (not $tab) }}{{ warnf "filetree: tab label must not be empty at %s; ignoring tab" .Position }}{{ end -}}
{{- $group := index $policy.consumed "group" | default "" -}}
{{- $value := index $policy.consumed "value" | default "" -}}
{{- if and (or (isset $policy.consumed "group") (isset $policy.consumed "value")) (not $tab) }}{{ warnf "filetree: group/value require tab at %s; ignoring them" .Position }}{{ end -}}
{{- $inner := replace (printf "%s" .Inner) "\r" "" -}}
{{- if not (strings.ContainsNonSpace $inner) }}{{ warnf "filetree: the fence requires tree entries at %s; rendering nothing" .Position }}{{ end -}}
{{- $parsed := partial "content/filetree-parse.html" (dict "source" $inner "position" .Position) -}}
{{- $nodes := $parsed.nodes -}}
{{- $static := in (slice "print" "rss") $format -}}
{{- $html := "" -}}

{{- if eq $format "markdown" -}}
  {{- /* re-emit the fence so the Markdown output stays valid source */ -}}
  {{- $attrText := "" -}}
  {{- with $title }}{{ $attrText = printf ` {title=%q}` . }}{{ end -}}
  {{- $html = printf "```filetree%s\n%s```\n" $attrText (strings.TrimRight "\n" $inner | printf "%s\n") -}}
{{- else if eq $format "rss" -}}
  {{- $html = printf `
%s
` (htmlEscape (strings.TrimRight "\n" $inner)) -}}
{{- else -}}
  {{- $id := printf "td-filetree-%s-f%d" (md5 $page.RelPermalink) .Ordinal -}}
  {{- $classes := slice "td-filetree" -}}
  {{- if $static }}{{ $classes = $classes | append "td-filetree--static" }}{{ end -}}
  {{- if not $parsed.hasComments }}{{ $classes = $classes | append "td-filetree--plain" }}{{ end -}}
  {{- with $policy.class }}{{ $classes = $classes | append . }}{{ end -}}
  {{- $count := len $nodes -}}
  {{- $rows := slice -}}
  {{- range $i, $node := $nodes -}}
    {{- $icons := partial "content/filetree-icon.html" (dict "name" $node.name "kind" $node.kind "icon" $node.icon) -}}
    {{- $iconClass := printf "td-filetree__icon td-filetree__icon--%s" $node.kind -}}
    {{- with $node.tone }}{{ $iconClass = printf "%s td-filetree__icon--%s" $iconClass . }}{{ end -}}
    {{- $iconHTML := "" -}}
    {{- if eq (len $icons) 2 -}}
      {{- $iconHTML = printf `` $iconClass (index $icons 0) (index $icons 1) -}}
    {{- else -}}
      {{- $iconHTML = printf `` $iconClass (index $icons 0) -}}
    {{- end -}}
    {{- $nameHTML := htmlEscape $node.name -}}
    {{- if $node.href -}}
      {{- $nameHTML = printf `%s` (htmlEscape $node.href) (cond $node.external ` rel="external"` "") $nameHTML -}}
    {{- end -}}
    {{- $commentHTML := "" -}}
    {{- if $parsed.hasComments -}}
      {{- if $node.comment -}}
        {{- $commentHTML = printf `#%s` (htmlEscape $node.comment) (htmlEscape $node.comment) -}}
      {{- else -}}
        {{- $commentHTML = `` -}}
      {{- end -}}
    {{- end -}}
    {{- $row := printf `%s%s%s` $node.depth $iconHTML (htmlEscape $node.name) $nameHTML $commentHTML -}}
    {{- $nextDepth := 0 -}}
    {{- if lt (add $i 1) $count }}{{ $nextDepth = index (index $nodes (add $i 1)) "depth" }}{{ end -}}
    {{- $li := "" -}}
    {{- if $node.children -}}
      {{- if $static -}}
        {{- $li = printf `
  • %s
      ` $node.kind $row -}}
      {{- else -}}
        {{- $li = printf `
    • %s
        ` $node.kind (cond $node.open " open" "") $row -}}
      {{- end -}}
    {{- else -}}
      {{- $li = printf `
      • %s
        • ` $node.kind $row -}}
    {{- end -}}
    {{- /* close every level the next entry steps out of */ -}}
    {{- $closes := sub $node.depth $nextDepth -}}
    {{- if $node.children }}{{ $closes = sub (add $node.depth 1) $nextDepth }}{{ end -}}
    {{- if gt $closes 0 -}}
      {{- range seq $closes -}}
        {{- if $static }}{{ $li = printf "%s
      • " $li }}{{ else }}{{ $li = printf "%s
    • " $li }}{{ end -}} {{- end -}} {{- end -}} {{- $rows = $rows | append $li -}} {{- end -}} {{- $nameCol := printf "%.1fch" $parsed.nameWidth -}} {{- $resizable := and (not $static) $parsed.hasComments -}} {{- $genericAttrs := "" -}} {{- range $name, $value := $policy.generic }}{{ $genericAttrs = printf `%s %s="%s"` $genericAttrs $name (htmlEscape $value) }}{{ end -}} {{- $html = printf `
    ` (delimit $classes " ") $nameCol (cond (ne $title "") (printf ` role="group" aria-labelledby="%s-title"` $id) "") (cond $resizable " data-td-filetree" "") $genericAttrs -}} {{- with $title -}} {{- $html = printf `%s

    %s

    ` $html $id (htmlEscape .) -}} {{- end -}} {{- $divider := "" -}} {{- if $resizable -}} {{- $page.Store.Set "hasFileTree" true -}} {{- $divider = printf `` (htmlEscape (T "ui_filetree_divider")) -}} {{- end -}} {{- $html = printf `%s
    %s
      %s
    ` $html $divider (delimit $rows "") -}} {{- end -}} {{- if $tab -}} {{- partial "content/tab-block.html" (dict "page" $page "position" .Position "tab" $tab "group" $group "value" $value "kind" "code" "content" ($html | safeHTML)) -}} {{- else -}} {{- $html | safeHTML -}} {{- end -}}
