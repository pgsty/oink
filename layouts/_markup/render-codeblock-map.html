{{- /* ```map {caption="…" route=true zoom=15 height="420px"} — a YAML or JSON list of places. Each place: name (required), lat and lng (required WGS 84 numbers), text, image, link. The numbered place list is the content in every output; when params.map is complete, HTML adds an interactive map above it whose numbered markers match the list. Nothing is fetched at build time; tiles load in the reader's browser only on sites that configured a tile service. */ -}} {{- $page := .PageInner | default .Page -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- $position := .Position -}} {{- $policy := partial "content/attributes.html" (dict "attributes" .Attributes "allowed" (slice "id" "caption" "route" "zoom" "height" "tab" "group" "value") "position" $position "kind" "map" ) -}} {{- $caption := strings.TrimSpace (index $policy.consumed "caption" | default "") -}} {{- $route := false -}} {{- with index $policy.consumed "route" -}} {{- $raw := lower (strings.TrimSpace .) -}} {{- if eq $raw "true" }}{{ $route = true }}{{ else if ne $raw "false" }}{{ warnf "map: route must be true or false; got %q at %s; using false" . $position }}{{ end -}} {{- end -}} {{- $zoom := 0 -}} {{- with index $policy.consumed "zoom" -}} {{- if and (findRE `^[1-9][0-9]?$` .) (ge (int .) 1) (le (int .) 22) -}} {{- $zoom = int . -}} {{- else -}} {{- warnf "map: zoom must be a whole number from 1 to 22; got %q at %s; fitting the places instead" . $position -}} {{- end -}} {{- end -}} {{- $height := "" -}} {{- with index $policy.consumed "height" -}} {{- $height = partial "validate.html" (dict "value" . "kind" "length" "fallback" "" "key" "map height" "where" $position) -}} {{- /* The figure has no height of its own, so a percentage resolves to nothing, and a zero length leaves a canvas no reader can see. */ -}} {{- if and $height (or (strings.HasSuffix $height "%") (not (findRE `[1-9]` $height))) -}} {{- warnf "map: height must be a positive px, rem, em, vh or vw length; got %q at %s; using the default" $height $position -}} {{- $height = "" -}} {{- end -}} {{- end -}} {{- $tab := strings.TrimSpace (index $policy.consumed "tab" | default "") -}} {{- if and (isset $policy.consumed "tab") (not $tab) }}{{ warnf "map: tab label must not be empty at %s; ignoring tab" $position }}{{ end -}} {{- if and (or (isset $policy.consumed "group") (isset $policy.consumed "value")) (not $tab) }}{{ warnf "map: group/value require tab at %s; ignoring them" $position }}{{ end -}} {{- /* Places. */ -}} {{- $source := strings.TrimSpace .Inner -}} {{- $items := slice -}} {{- $parsed := false -}} {{- if not $source -}} {{- warnf "map: the fence requires a list of places at %s; rendering nothing" $position -}} {{- else -}} {{- /* Parse as YAML by name, never by content sniffing: a YAML list such as `- {name: …}` would otherwise be read as JSON. JSON is YAML. */ -}} {{- $result := try (resources.FromString (printf "td-map/%s.yaml" (md5 $source)) $source | transform.Unmarshal) -}} {{- if $result.Err -}} {{- warnf "map: places are not valid YAML/JSON at %s: %s; rendering nothing" $position $result.Err -}} {{- else if not (reflect.IsSlice $result.Value) -}} {{- warnf "map: the fence must hold a list of places at %s; rendering nothing" $position -}} {{- else -}} {{- $items = $result.Value -}} {{- $parsed = true -}} {{- end -}} {{- end -}} {{- $known := slice "name" "lat" "lng" "text" "image" "link" -}} {{- $numeric := slice "float64" "int" "int64" "uint64" -}} {{- $places := slice -}} {{- range $index, $item := $items -}} {{- $n := add $index 1 -}} {{- if not (reflect.IsMap $item) -}} {{- warnf "map: place %d must be a mapping at %s; skipping it" $n $position -}} {{- continue -}} {{- end -}} {{- range $key, $_ := $item -}} {{- if not (in $known $key) }}{{ warnf "map: place %d has unknown field %q at %s (allowed: %s); ignoring it" $n $key $position (delimit $known ", ") }}{{ end -}} {{- end -}} {{- $name := "" -}} {{- $rawName := index $item "name" -}} {{- if ne $rawName nil -}} {{- if eq (printf "%T" $rawName) "string" -}} {{- $name = strings.TrimSpace $rawName -}} {{- else -}} {{- warnf "map: place %d name must be a string; got %T at %s; skipping it" $n $rawName $position -}} {{- continue -}} {{- end -}} {{- end -}} {{- if not $name -}} {{- warnf "map: place %d requires a name at %s; skipping it" $n $position -}} {{- continue -}} {{- end -}} {{- $lat := index $item "lat" -}} {{- $lng := index $item "lng" -}} {{- if not (and (in $numeric (printf "%T" $lat)) (in $numeric (printf "%T" $lng))) -}} {{- warnf "map: place %q requires numeric lat and lng at %s; skipping it" $name $position -}} {{- continue -}} {{- end -}} {{- $latF := float $lat -}} {{- $lngF := float $lng -}} {{- /* Negated closed-interval check rejects NaN alongside ±Inf and out-of-range numbers, because every IEEE 754 comparison with NaN is false. */ -}} {{- if not (and (ge $latF -90.0) (le $latF 90.0) (ge $lngF -180.0) (le $lngF 180.0)) -}} {{- warnf "map: place %q has lat %v and lng %v outside -90..90 and -180..180 at %s; skipping it" $name $lat $lng $position -}} {{- continue -}} {{- end -}} {{- $latStr := strings.TrimRight "." (strings.TrimRight "0" (printf "%.6f" $latF)) -}} {{- if eq $latStr "-0" }}{{ $latStr = "0" }}{{ end -}} {{- $lngStr := strings.TrimRight "." (strings.TrimRight "0" (printf "%.6f" $lngF)) -}} {{- if eq $lngStr "-0" }}{{ $lngStr = "0" }}{{ end -}} {{- $text := "" -}} {{- $rawText := index $item "text" -}} {{- if ne $rawText nil -}} {{- if eq (printf "%T" $rawText) "string" -}} {{- $text = strings.TrimSpace $rawText -}} {{- else -}} {{- warnf "map: place %q text must be a string; got %T at %s; ignoring it" $name $rawText $position -}} {{- end -}} {{- end -}} {{- $thumb := dict -}} {{- with index $item "image" -}} {{- $image := partial "content/image-resolve.html" (dict "name" "map" "label" "map fence" "position" $position "page" $page "src" . "decorative" true) -}} {{- if $image.src -}} {{- $src := $image.src -}}{{- $w := $image.width -}}{{- $h := $image.height -}} {{- if $image.processable -}} {{- with $image.resource.Fit "480x360" }}{{ $src = .RelPermalink }}{{ $w = .Width }}{{ $h = .Height }}{{ end -}} {{- end -}} {{- if eq $format "rss" }}{{ $src = absURL $src }}{{ end -}} {{- $thumb = dict "src" $src "width" $w "height" $h -}} {{- end -}} {{- end -}} {{- $href := "" -}} {{- $hrefExternal := false -}} {{- with index $item "link" -}} {{- $link := partial "content/url.html" (dict "name" "map" "label" "map fence" "parameter" (printf "place %d link" $n) "position" $position "value" . "kind" "link" "rssPage" (cond (eq $format "rss") $page nil)) -}} {{- $href = $link.href -}} {{- $hrefExternal = $link.external -}} {{- end -}} {{- $places = $places | append (dict "name" $name "lat" $latF "lng" $lngF "latStr" $latStr "lngStr" $lngStr "label" (printf "%s, %s" $latStr $lngStr) "text" $text "thumb" $thumb "href" $href "hrefExternal" $hrefExternal) -}} {{- end -}} {{- if and $parsed (not $places) }}{{ warnf "map: no valid place at %s; rendering nothing" $position }}{{ end -}} {{- if $places -}} {{- $config := partialCached "content/map-config.html" . $page.Site.Language.Lang -}} {{- $interactive := and $config.enable (eq $format "html") -}} {{- $id := printf "td-map-%s-f%d" (md5 $page.RelPermalink) .Ordinal -}} {{- with index $policy.consumed "id" -}} {{- with partial "book/validate-id.html" (dict "name" "map" "label" "map fence" "value" . "position" $position) }}{{ $id = . }}{{ end -}} {{- end -}} {{- $classes := slice "td-map" -}} {{- if $interactive }}{{ $classes = $classes | append "td-map--interactive" }}{{ end -}} {{- $classes = $classes | append $policy.classes -}} {{- $rows := slice -}} {{- range $place := $places -}} {{- $osm := printf "https://www.openstreetmap.org/?mlat=%s&mlon=%s#map=16/%s/%s" .latStr .lngStr .latStr .lngStr -}} {{- $nameHTML := htmlEscape .name -}} {{- with .href }}{{ $nameHTML = printf `%s` (htmlEscape .) (cond $place.hrefExternal ` rel="external"` "") $nameHTML }}{{ end -}} {{- $thumbHTML := "" -}} {{- with $place.thumb.src -}} {{- $thumbHTML = printf `` (htmlEscape .) (cond (gt ($place.thumb.width | default 0) 0) (printf ` width="%d"` $place.thumb.width) "") (cond (gt ($place.thumb.height | default 0) 0) (printf ` height="%d"` $place.thumb.height) "") -}} {{- end -}} {{- $row := printf `
  • %s%s %s%s
    • ` $nameHTML (cond (ne .text "") (printf ` %s` (htmlEscape .text)) "") (htmlEscape $osm) (htmlEscape (T "ui_map_open_osm")) (htmlEscape .label) $thumbHTML -}} {{- $rows = $rows | append $row -}} {{- end -}} {{- $html := "" -}} {{- $canvas := "" -}} {{- $data := "" -}} {{- if $interactive -}} {{- $page.Store.Set "hasMap" true -}} {{- $canvas = printf `
    ` (htmlEscape (cond (ne $caption "") $caption (T "ui_map_label"))) (cond (ne $height "") (printf ` style="height: %s"` $height) "") -}} {{- $json := slice -}} {{- range $places -}} {{- $entry := dict "name" .name "lat" .lat "lng" .lng "text" .text "thumb" (.thumb.src | default "") "href" .href -}} {{- if and .thumb.src (gt (.thumb.width | default 0) 0) (gt (.thumb.height | default 0) 0) -}} {{- $entry = merge $entry (dict "thumbWidth" .thumb.width "thumbHeight" .thumb.height) -}} {{- end -}} {{- $json = $json | append $entry -}} {{- end -}} {{- $data = printf `` (dict "places" $json "route" $route "zoom" $zoom | jsonify) -}} {{- end -}} {{- $html = printf `` $id (delimit $classes " ") (cond $interactive " data-td-map" "") (partial "content/attrs.html" $policy.generic) $canvas (delimit $rows "") (cond (ne $caption "") (printf `` (htmlEscape $caption)) "") $data -}} {{- if $tab -}} {{- partial "content/tab-block.html" (dict "page" $page "position" $position "tab" $tab "group" (index $policy.consumed "group" | default "") "value" (index $policy.consumed "value" | default "") "kind" "code" "content" ($html | safeHTML) ) -}} {{- else -}} {{- $html | safeHTML -}} {{- end -}} {{- end -}}
