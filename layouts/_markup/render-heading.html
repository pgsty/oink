{{- /* Every heading carries its id and a self-link anchor that the content styles reveal on hover (`.td-heading-self-link`). Block attributes follow the shared policy: `id` is consumed, `class` passes through validated, data-* / aria-* pass through, and unsupported attributes warn and drop. The `.update` author marker makes the heading a changelog entry: it also accepts string `date` (valid calendar YYYY, YYYY-MM or YYYY-MM-DD) and `tags` (comma-separated plain text), rendered as a metadata line after the heading so the heading text, its TOC entry, anchor and search title stay exactly as authored. Its items are separated by in-flow `, ` text that the styles shrink to nothing, so plain-text copy, search, feeds and screen readers keep them apart. Goldmark gives every heading an `id` attribute. A heading whose only attribute is that id needs no policy, so it skips both partials and emits the same bytes the full path would. */ -}} {{- if and (eq (len .Attributes) 1) (eq (index .Attributes "id") .Anchor) -}} {{- /* Goldmark drops a whole attribute block it cannot parse, such as an unquoted date=2026-11-01, and leaves it in the heading text. */ -}} {{- if and (strings.Contains .PlainText "{") (findRE `\{\s*\.update\b[^{}]*\}\s*$` .PlainText 1) -}} {{- warnf "heading %q: its {.update …} attributes were not parsed, so they stay in the heading text; quote every value, as in date=\"2026-11-01\", in %s" .PlainText .Page.Path -}} {{- end -}} {{- .Text -}} {{ else -}} {{- $allowed := slice "id" -}} {{- $update := false -}} {{- with index .Attributes "class" -}} {{- if in (findRE `\S+` (printf "%v" .)) "update" -}} {{- $update = true -}} {{- $allowed = slice "id" "date" "tags" -}} {{- end -}} {{- end -}} {{- $policy := partial "content/attributes.html" (dict "attributes" .Attributes "allowed" $allowed "position" .Page.Path "kind" "heading") -}} {{- $id := index $policy.consumed "id" | default .Anchor -}} {{- .Text -}} {{ if $update -}} {{- $date := "" -}} {{- $rawDate := index .Attributes "date" -}} {{- if ne $rawDate nil -}} {{- if ne (printf "%T" $rawDate) "string" -}} {{- warnf "heading %q: update date must be YYYY, YYYY-MM or YYYY-MM-DD (string); got %T in %s; dropping the date" .PlainText $rawDate .Page.Path -}} {{- else -}} {{- $candidate := strings.TrimSpace $rawDate -}} {{- if $candidate -}} {{- $validDate := findRE `^[0-9]{4}(-(0[1-9]|1[0-2])(-(0[1-9]|[12][0-9]|3[01]))?)?$` $candidate -}} {{- if and $validDate (eq (len $candidate) 10) -}} {{- if (try (time.AsTime $candidate)).Err -}} {{- $validDate = false -}} {{- end -}} {{- end -}} {{- if $validDate -}} {{- $date = $candidate -}} {{- else -}} {{- warnf "heading %q: update date must be YYYY, YYYY-MM or YYYY-MM-DD (valid calendar date); got %q in %s; dropping the date" .PlainText $candidate .Page.Path -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $tags := slice -}} {{- $rawTags := index .Attributes "tags" -}} {{- if ne $rawTags nil -}} {{- if ne (printf "%T" $rawTags) "string" -}} {{- warnf "heading %q: update tags must be a string; got %T in %s; dropping tags" .PlainText $rawTags .Page.Path -}} {{- else -}} {{- range split (replace $rawTags "，" ",") "," -}} {{- with strings.TrimSpace . }}{{ $tags = $tags | append . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- if or $date $tags -}} {{- $sep := false -}}
{{- with $date }}{{ . }}{{ $sep = true }}{{ end -}} {{- range $tags }}{{ if $sep }}, {{ end }}{{ . }}{{ $sep = true }}{{ end -}}
{{ end -}} {{- end -}} {{- end -}}
