{{- /* Passthrough (delimiter math) render hook. Block math followed by an attribute line becomes a numbered Book equation (v5 §3.12): $$ X \approx \frac{C}{R+Z} $$ {#eq_x num="5.3" caption="…"} Inline math never carries attributes. */ -}} {{- $page := .PageInner | default .Page -}} {{- $isBlock := ne .Type "inline" -}} {{- $attrs := .Attributes | default dict -}} {{- if and $isBlock (gt (len $attrs) 0) -}} {{- $policy := partial "content/attributes.html" (dict "attributes" $attrs "allowed" (slice "id" "num" "caption") "position" .Position "kind" "math") -}} {{- $numRaw := index $policy.consumed "num" | default "" -}} {{- $idRaw := index $policy.consumed "id" | default "" -}} {{- $caption := strings.TrimSpace (index $policy.consumed "caption" | default "") -}} {{- if not $numRaw }}{{ errorf "math: id/caption attributes require num at %s" .Position }}{{ end -}} {{- $num := partial "book/validate-num.html" (dict "name" "math" "value" $numRaw "position" .Position) -}} {{- $id := printf "eq-%s" $num -}} {{- if $idRaw }}{{ $id = partial "book/validate-id.html" (dict "name" "math" "value" $idRaw "position" .Position) }}{{ end -}} {{- partial "book/register-target.html" (dict "page" $page "kind" "eq" "num" $num "id" $id "caption" $caption "position" .Position "ordinal" .Ordinal) -}} {{- $label := partial "book/label.html" (dict "kind" "eq" "num" $num) -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- if in (slice "markdown" "rss") $format -}} {{- printf "

%s.%s

" $label (cond (ne $caption "") (printf " %s" (htmlEscape $caption)) "") | safeHTML -}} {{- partial "scripts/math.html" . -}} {{- else -}} {{- end -}} {{- else -}} {{ partial "scripts/math.html" . -}} {{- end -}}
