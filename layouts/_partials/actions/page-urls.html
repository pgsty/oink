{{- /* Resolve page-action destinations once for descriptors and HTML. */ -}} {{- $p := . -}} {{- $markdownURL := "" -}} {{- with $p.AlternativeOutputFormats.Get "markdown" }}{{ $markdownURL = .RelPermalink }}{{ end -}} {{- $printURL := "" -}} {{- with $p.CurrentSection -}} {{- with .AlternativeOutputFormats.Get "print" }}{{ $printURL = .RelPermalink }}{{ end -}} {{- end -}} {{- $editURL := "" -}} {{- $historyURL := "" -}} {{- $childURL := "" -}} {{- $issueURL := "" -}} {{- $projectIssueURL := "" -}} {{- $repoPath := "" -}} {{- $repo := $p.Param "github_repo" -}} {{- $projectRepo := $p.Param "github_project_repo" | default $repo | default "" -}} {{- if and $p.File $repo -}} {{- $sourcePath := strings.TrimPrefix (add hugo.WorkingDir "/") $p.File.Filename -}} {{- $subdir := $p.Param "github_subdir" | default "" -}} {{- $branch := $p.Param "github_branch" | default "main" -}} {{- $pathBase := $p.Param "path_base_for_github_subdir" -}} {{- $pathRename := "" -}} {{- if reflect.IsMap $pathBase -}} {{- $pathRename = $pathBase.to -}} {{- $pathBase = $pathBase.from -}} {{- end -}} {{- with $pathBase }}{{ $sourcePath = replaceRE . $pathRename $sourcePath }}{{ end -}} {{- $repoPath = printf "%s/%s/%s" $branch $subdir $sourcePath | replaceRE "//+" "/" -}} {{- $editURL = printf "%s/edit/%s" $repo $repoPath -}} {{- /* Same derivation as the edit link, so a repository that supports one supports the other. */ -}} {{- $historyURL = printf "%s/commits/%s" $repo $repoPath -}} {{- $issueURL = printf "%s/issues/new?%s" $repo (querify "title" $p.Title) -}} {{- $stub := resources.Get "stubs/new-page-template.md" -}} {{- $childQS := querify "value" $stub.Content "filename" "change-me.md" -}} {{- $childURL = printf "%s/new/%s?%s" $repo (path.Dir $repoPath) $childQS -}} {{- end -}} {{- /* A project issue is filed against the product, not against the page, so it needs no source path and no `github_repo`. Nesting it with the page-derived links meant a site that declared only a project repository -- the common shape for a docs site whose code lives elsewhere -- lost the one action it had configured. */ -}} {{- with $p.Param "github_project_repo" }}{{ $projectIssueURL = printf "%s/issues/new" . }}{{ end -}} {{- return (dict "markdown" $markdownURL "printSection" $printURL "edit" $editURL "history" $historyURL "createChild" $childURL "createIssue" $issueURL "createProjectIssue" $projectIssueURL "projectRepo" $projectRepo "repoPath" $repoPath ) -}}
