{{- $page := .page -}} {{- $id := .id -}} {{- $num := .num -}} {{- $kind := .kind -}} {{- $position := .position | default $page.Path -}} {{- $ordinal := .ordinal | default 0 -}} {{- /* Document order comes from the source position (file:line:col) so shortcode targets and render-hook targets (native forms) interleave in one list instead of one form sorting ahead of the other; the ordinal is the fallback. Hugo renders a position as a quoted string, so the quotes come off before matching. Identity stays keyed on the ordinal: every output format shares one Page Store and only the first pass registers, but a print aggregate renders the page from a string, where the same target reports a line number short by the front matter. Ordinals are stable across those passes. */ -}} {{- $identity := printf "%s:%d" $page.Path $ordinal -}} {{- $orderKey := printf "%08d-00000" $ordinal -}} {{- with findRESubmatch `:(\d+):(\d+)$` (trim (printf "%v" $position) "\"") -}} {{- $m := index . 0 -}} {{- $orderKey = printf "%08d-%05d" (int (index $m 1)) (int (index $m 2)) -}} {{- end -}} {{- $targets := $page.Store.Get "tdBookTargets" | default (dict) -}} {{- if isset $targets $id -}} {{- $existing := index $targets $id -}} {{- if not (and (eq $existing.identity $identity) (eq $existing.kind $kind) (eq $existing.num $num)) -}} {{- errorf "book component: duplicate id %q at %s (already used by %s %s at %s)" $id $position $existing.kind $existing.num $existing.position -}} {{- end -}} {{- end -}} {{- $numbers := $page.Store.Get "tdBookTargetNums" | default (dict) -}} {{- $numberKey := printf "%s:%s" $kind $num -}} {{- if and (not (isset $targets $id)) (isset $numbers $numberKey) -}} {{- $existing := index $numbers $numberKey -}} {{- if ne $existing.id $id }}{{ errorf "book component: duplicate %s number %q at %s (already uses id %q)" $kind $num $position $existing.id }}{{ end -}} {{- end -}} {{- if not (isset $targets $id) -}} {{- $entry := dict "id" $id "num" $num "kind" $kind "caption" (.caption | default "") "ordinal" $ordinal "identity" $identity "page" $page "position" $position -}} {{- $page.Store.SetInMap "tdBookTargets" $id $entry -}} {{- $page.Store.SetInMap "tdBookTargetNums" $numberKey $entry -}} {{- $page.Store.SetInMap "tdBookTargetOrder" (printf "%s-%s" $orderKey $id) $entry -}} {{- end -}}
