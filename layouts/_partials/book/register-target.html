{{- $page := .page -}} {{- $id := .id -}} {{- $num := .num -}} {{- $kind := .kind -}} {{- $position := .position | default $page.Path -}} {{- $ordinal := .ordinal | default 0 -}} {{- $source := .source | default "shortcode" -}} {{- $block := hugo.Context.MarkupScope -}} {{- if not (hasPrefix $block "td-block/") }}{{ $block = "" }}{{ end -}} {{- $instance := printf "%d" $ordinal -}} {{- with .shortcode }}{{ $instance = partial "content/shortcode-ordinal.html" . }}{{ end -}} {{- $identity := printf "%s:%s:%s:%s" $page.Path $block $source $instance -}} {{- $orderKey := printf "%08d-00000" $ordinal -}} {{- with findRESubmatch `:(\d+):(\d+)$` (trim (printf "%v" $position) "\"") -}} {{- $m := index . 0 -}} {{- $orderKey = printf "%08d-%05d" (int (index $m 1)) (int (index $m 2)) -}} {{- end -}} {{- $targets := $page.Store.Get "tdBookTargets" | default (dict) -}} {{- $accepted := true -}} {{- if isset $targets $id -}} {{- $existing := index $targets $id -}} {{- if not (and (eq $existing.identity $identity) (eq $existing.kind $kind) (eq $existing.num $num)) -}} {{- warnf "book component: duplicate id %q at %s (already used by %s %s at %s); keeping the first" $id $position $existing.kind $existing.num $existing.position -}} {{- $accepted = false -}} {{- end -}} {{- end -}} {{- $numbers := $page.Store.Get "tdBookTargetNums" | default (dict) -}} {{- $numberKey := printf "%s:%s" $kind $num -}} {{- if and (not (isset $targets $id)) (isset $numbers $numberKey) -}} {{- $existing := index $numbers $numberKey -}} {{- if ne $existing.id $id }}{{ warnf "book component: duplicate %s number %q at %s (already uses id %q); keeping the first" $kind $num $position $existing.id }}{{ end -}} {{- end -}} {{- if not (isset $targets $id) -}} {{- $entry := dict "id" $id "num" $num "kind" $kind "caption" (.caption | default "") "ordinal" $ordinal "identity" $identity "page" $page "position" $position -}} {{- $page.Store.SetInMap "tdBookTargets" $id $entry -}} {{- if not (isset $numbers $numberKey) -}} {{- $page.Store.SetInMap "tdBookTargetNums" $numberKey $entry -}} {{- end -}} {{- $page.Store.SetInMap "tdBookTargetOrder" (printf "%s-%s" $orderKey $id) $entry -}} {{- end -}} {{- return $accepted -}}
