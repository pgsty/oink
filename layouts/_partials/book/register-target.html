{{- $page := .page -}} {{- $id := .id -}} {{- $num := .num -}} {{- $kind := .kind -}} {{- $position := .position | default $page.Path -}} {{- $ordinal := .ordinal | default 0 -}} {{- /* Identity and document order come from the source position (file:line:col) so shortcode targets and render-hook targets (native forms) share one namespace and one ordering; ordinal is the fallback. */ -}} {{- $identity := printf "%s:%d" $page.Path $ordinal -}} {{- $orderKey := printf "%08d-00000" $ordinal -}} {{- with findRESubmatch `:(\d+):(\d+)$` (printf "%v" $position) -}} {{- $m := index . 0 -}} {{- $identity = printf "%s@%s:%s" $page.Path (index $m 1) (index $m 2) -}} {{- $orderKey = printf "%08d-%05d" (int (index $m 1)) (int (index $m 2)) -}} {{- end -}} {{- $targets := $page.Store.Get "tdBookTargets" | default (dict) -}} {{- if isset $targets $id -}} {{- $existing := index $targets $id -}} {{- if not (and (eq $existing.identity $identity) (eq $existing.kind $kind) (eq $existing.num $num)) -}} {{- errorf "book component: duplicate id %q at %s (already used by %s %s at %s)" $id $position $existing.kind $existing.num $existing.position -}} {{- end -}} {{- end -}} {{- $numbers := $page.Store.Get "tdBookTargetNums" | default (dict) -}} {{- $numberKey := printf "%s:%s" $kind $num -}} {{- if and (not (isset $targets $id)) (isset $numbers $numberKey) -}} {{- $existing := index $numbers $numberKey -}} {{- if ne $existing.id $id }}{{ errorf "book component: duplicate %s number %q at %s (already uses id %q)" $kind $num $position $existing.id }}{{ end -}} {{- end -}} {{- if not (isset $targets $id) -}} {{- $entry := dict "id" $id "num" $num "kind" $kind "caption" (.caption | default "") "ordinal" $ordinal "identity" $identity "page" $page "position" $position -}} {{- $page.Store.SetInMap "tdBookTargets" $id $entry -}} {{- $page.Store.SetInMap "tdBookTargetNums" $numberKey $entry -}} {{- $page.Store.SetInMap "tdBookTargetOrder" (printf "%s-%s" $orderKey $id) $entry -}} {{- end -}}
