{{- $shortcode := .shortcode -}} {{- $allowed := .allowed -}} {{- if and (gt (len $shortcode.Params) 0) (not $shortcode.IsNamedParams) -}} {{- errorf "shortcode %q accepts named parameters only at %s" $shortcode.Name $shortcode.Position -}} {{- end -}} {{- range $name, $_ := $shortcode.Params -}} {{- if not (in $allowed $name) -}} {{- errorf "shortcode %q: unsupported parameter %q at %s" $shortcode.Name $name $shortcode.Position -}} {{- end -}} {{- end -}} {{- return "" -}}
