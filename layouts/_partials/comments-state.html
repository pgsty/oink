{{/* Hugo normalizes keys under Params to lowercase. Keep all map lookups in this helper lowercase so comments.html and the page layouts share one source of truth for activation and legacy Disqus fallback. */ -}} {{- /* params.comments is a map (enable, type, giscus) or the bare boolean shorthand `comments: true|false`. */ -}} {{- $raw := .Site.Params.comments -}} {{- $comments := dict -}} {{- $enabledValue := false -}} {{- if reflect.IsMap $raw -}} {{- $comments = $raw -}} {{- $enabledValue = index $comments "enable" | default false -}} {{- else if ne $raw nil -}} {{- $enabledValue = $raw -}} {{- end -}} {{- $pageOverride := isset .Params "comments" -}} {{- if $pageOverride -}} {{- $enabledValue = .Params.comments -}} {{- end -}} {{- $enabledText := printf "%v" $enabledValue | lower -}} {{- $enabled := or (eq $enabledText "true") (eq $enabledText "1") -}} {{- $commentType := index $comments "type" | default "" | lower -}} {{- $giscus := index $comments "giscus" | default dict -}} {{- $missing := slice -}} {{- $required := slice (dict "key" "repo" "label" "repo") (dict "key" "repoid" "label" "repoId") (dict "key" "category" "label" "category") (dict "key" "categoryid" "label" "categoryId") -}} {{- range $required -}} {{- $value := index $giscus .key | default "" -}} {{- if not (strings.TrimSpace (printf "%v" $value)) -}} {{- $missing = $missing | append .label -}} {{- end -}} {{- end -}} {{- $active := and $enabled (eq $commentType "giscus") (eq (len $missing) 0) -}} {{- $suppressDisqus := or $active (and $pageOverride (not $enabled)) -}} {{- return dict "enabled" $enabled "type" $commentType "missing" $missing "active" $active "suppressDisqus" $suppressDisqus -}}
