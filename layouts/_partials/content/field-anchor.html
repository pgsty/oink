{{- /* Allocate a stable, page-unique anchor id for one Fields entry so both the `{.fields}` table hook and the `fields`/`field` shortcode pair produce linkable definitions. Input dict: page the page the entry belongs to name the field name (plain text) identity caller-stable identity of this entry inside the page, e.g. "tb3#2" (table ordinal / row) or "sc1#4" (shortcode ordinal / child). Re-registering the same identity is idempotent, so a second render pass over the same document keeps its ids. The registry lives on the page store, keyed by output format, because Hugo renders one page once per format and the store outlives a single pass. A name that collides inside one page takes the next free `-2`, `-3`, … suffix, the way Goldmark disambiguates repeated headings. Returns the id (never empty). */ -}} {{- $page := .page -}} {{- $identity := printf "%v" .identity -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- /* Field names are identifiers, not prose. Goldmark's heading rule (`anchorize`) deletes punctuation instead of converting it, which turns `params.ui.typography` into `paramsuitypography` — an id no reader can predict, and therefore one no reader can link to. Lowercase, then every run of punctuation collapses to a single hyphen, so the anchor stays derivable from the name by eye: `params.ui.typography` -> `field-params-ui-typography`, `--dry-run` -> `field-dry-run`. `_` counts as a word character because configuration keys carry meaning in it, and Unicode letters survive, so CJK and Arabic names keep reading as themselves. */ -}} {{- $slug := strings.Trim (replaceRE `[^\p{L}\p{N}_]+` "-" (lower (printf "%v" .name))) "-" -}} {{- if not $slug }}{{ $slug = "entry" }}{{ end -}} {{- $base := printf "field-%s" $slug -}} {{- $registry := $page.Store.Get "tdFieldAnchors" | default (dict) -}} {{- $id := "" -}} {{- range $i := seq 1 64 -}} {{- if not $id -}} {{- $candidate := cond (eq $i 1) $base (printf "%s-%d" $base $i) -}} {{- $key := printf "%s|%s" $format $candidate -}} {{- $taken := printf "%v" (index $registry $key | default "") -}} {{- if or (not $taken) (eq $taken $identity) -}} {{- $id = $candidate -}} {{- $page.Store.SetInMap "tdFieldAnchors" $key $identity -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $id -}} {{- /* Past 64 entries sharing one name, fall back to the caller identity, which is unique inside the page by construction, so the document still has no duplicate id. Warn rather than stop the build: the page is readable, only these anchors stop being guessable. */ -}} {{- $id = printf "%s-%s" $base (strings.Trim (replaceRE `[^\p{L}\p{N}_]+` "-" (lower $identity)) "-") -}} {{- warnf "fields: more than 64 entries on %s share the name %q; %q falls back to a position-derived anchor" $page.Path .name $id -}} {{- $page.Store.SetInMap "tdFieldAnchors" (printf "%s|%s" $format $id) $identity -}} {{- end -}} {{- return $id -}}
