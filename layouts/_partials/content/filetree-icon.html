Default folders expose both disclosure states; an authored icon deliberately replaces both. */ -}} {{- $kind := .kind -}} {{- if not (in (slice "folder" "file") $kind) -}} {{- errorf "internal FileTree icon kind %q is not supported" $kind -}} {{- end -}} {{- if .icon -}} {{- else if and (eq $kind "folder") .disclosure -}} {{- else if eq $kind "folder" -}} {{- else -}} {{- end }}
