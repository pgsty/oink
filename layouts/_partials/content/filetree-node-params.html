{{- /* Validate and normalize the optional presentation and filesystem fields shared by FileTree folders and files. */ -}} {{- $shortcode := . -}} {{- $values := dict -}} {{- range $parameter := slice "icon" "color" "comment" "owner" "group" "mode" -}} {{- $value := "" -}} {{- if isset $shortcode.Params $parameter -}} {{- $parameterValue := $shortcode.Get $parameter -}} {{- if ne (printf "%T" $parameterValue) "string" -}} {{- errorf "shortcode %q: %s must be a string at %s" $shortcode.Name $parameter $shortcode.Position -}} {{- end -}} {{- $value = strings.TrimSpace $parameterValue -}} {{- if not $value -}} {{- errorf "shortcode %q: %s must not be empty at %s" $shortcode.Name $parameter $shortcode.Position -}} {{- end -}} {{- end -}} {{- $values = merge $values (dict $parameter $value) -}} {{- end -}} {{- with $values.icon -}} {{- if not (findRE `^fa-(solid|regular|brands) fa-[a-z0-9-]+$` .) -}} {{- errorf "shortcode %q: icon must be one Font Awesome class pair such as %q; got %q at %s" $shortcode.Name "fa-solid fa-file-code" . $shortcode.Position -}} {{- end -}} {{- end -}} {{- with $values.color -}} {{- $colors := slice "neutral" "primary" "secondary" "info" "success" "warning" "danger" -}} {{- if not (in $colors .) -}} {{- errorf "shortcode %q: color must be one of %s; got %q at %s" $shortcode.Name (delimit $colors " | ") . $shortcode.Position -}} {{- end -}} {{- end -}} {{- if and $values.group (not $values.owner) -}} {{- errorf "shortcode %q: group requires owner at %s" $shortcode.Name $shortcode.Position -}} {{- end -}} {{- return $values -}}
