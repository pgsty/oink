{{- /* Apply a validated Hugo image operation while preserving the full-size URL. */ -}} {{- $name := .name | default "image" -}} {{- /* Error label: the Markdown image hook and the gallery fence are not shortcodes, so their messages must not send authors looking for one. */ -}} {{- $label := printf "shortcode %q" $name -}} {{- if eq $name "image" }}{{ $label = "image" }}{{ else if eq $name "gallery" }}{{ $label = "gallery fence" }}{{ end -}} {{- $position := .position | default "unknown position" -}} {{- $image := .image -}} {{- $commandValue := .command -}} {{- if ne (printf "%T" $commandValue) "string" -}} {{- errorf "%s: command must be a string at %s" $label $position -}} {{- end -}} {{- $command := strings.TrimSpace $commandValue -}} {{- $commands := slice "Fit" "Resize" "Fill" "Crop" -}} {{- if not (in $commands $command) -}} {{- errorf "%s: command must be one of %s; got %q at %s" $label (delimit $commands " | ") $command $position -}} {{- end -}} {{- $optionsValue := .options -}} {{- if ne (printf "%T" $optionsValue) "string" -}} {{- errorf "%s: options must be a string at %s" $label $position -}} {{- end -}} {{- $options := strings.TrimSpace $optionsValue -}} {{- if not $options -}} {{- errorf "%s: options must not be empty at %s" $label $position -}} {{- end -}} {{- if not $image.processable -}} {{- errorf "%s: src %q resolved as %s and cannot be processed at %s" $label $image.source $image.kind $position -}} {{- end -}} {{- $result := false -}} {{- if eq $command "Fit" -}} {{- $result = try ($image.resource.Fit $options) -}} {{- else if eq $command "Resize" -}} {{- $result = try ($image.resource.Resize $options) -}} {{- else if eq $command "Fill" -}} {{- $result = try ($image.resource.Fill $options) -}} {{- else -}} {{- $result = try ($image.resource.Crop $options) -}} {{- end -}} {{- with $result.Err -}} {{- errorf "%s: %s %q failed for src %q at %s: %s" $label $command $options $image.source $position . -}} {{- end -}} {{- $rendered := $result.Value -}} {{- return (dict "src" $rendered.RelPermalink "fullSrc" $image.fullSrc "markdownSrc" (partial "content/markdown-url.html" $rendered.RelPermalink) "source" $image.source "kind" $image.kind "mediaType" $rendered.MediaType.Type "width" $rendered.Width "height" $rendered.Height "processable" true "processed" true "resource" $image.resource "renderedResource" $rendered "alt" $image.alt "decorative" $image.decorative "caption" $image.caption "byline" $image.byline ) -}}
