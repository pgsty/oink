{{- /* Resolve image metadata without fetching remote sources. */ -}} {{- $name := .name | default "image" -}} {{- /* Error label: the Markdown image hook and the gallery fence are not shortcodes, so their messages must not send authors looking for one. */ -}} {{- $label := printf "shortcode %q" $name -}} {{- if eq $name "image" }}{{ $label = "image" }}{{ else if eq $name "gallery" }}{{ $label = "gallery fence" }}{{ end -}} {{- $position := .position | default "unknown position" -}} {{- $page := .page -}} {{- $sourceValue := .src -}} {{- if ne (printf "%T" $sourceValue) "string" -}} {{- errorf "%s: src must be a string at %s" $label $position -}} {{- end -}} {{- $source := strings.TrimSpace $sourceValue -}} {{- if not $source -}} {{- errorf "%s: src must not be empty at %s" $label $position -}} {{- end -}} {{- $validated := partial "content/url.html" (dict "name" $name "parameter" "src" "position" $position "value" $source "kind" "image" ) -}} {{- $parsedResult := try (urls.Parse $source) -}} {{- with $parsedResult.Err -}} {{- errorf "%s: invalid src %q at %s: %s" $label $source $position . -}} {{- end -}} {{- $parsed := $parsedResult.Value -}} {{- $scheme := lower $parsed.Scheme -}} {{- $kind := "static" -}} {{- $resource := false -}} {{- $lookup := strings.TrimPrefix "/" $source -}} {{- if not $scheme -}} {{- if eq (.matchMode | default "exact") "legacy" -}} {{- $resource = $page.Resources.GetMatch (printf "**%s*" $lookup) -}} {{- else -}} {{- $resource = $page.Resources.Get $lookup -}} {{- end -}} {{- if $resource -}} {{- $kind = "page" -}} {{- else -}} {{- /* Branch-bundle resources: a regular page in a section may reference files that live beside it under the section directory (content/blog/pg/post/image.png → resource "post/image.png" of blog/pg). */ -}} {{- with $page.CurrentSection -}} {{- if ne . $page -}} {{- with .Resources.Get $lookup }}{{ $resource = . }}{{ $kind = "page" }}{{ end -}} {{- end -}} {{- end -}} {{- if not $resource -}} {{- $resource = resources.Get $lookup -}} {{- if $resource }}{{ $kind = "global" }}{{ end -}} {{- end -}} {{- end -}} {{- else -}} {{- $kind = "remote" -}} {{- end -}} {{- $src := $validated.href -}} {{- $fullSrc := $src -}} {{- $mediaType := "" -}} {{- $width := 0 -}} {{- $height := 0 -}} {{- $processable := false -}} {{- $resourceAlt := "" -}} {{- $byline := "" -}} {{- if $resource -}} {{- if not (reflect.IsImageResource $resource) -}} {{- errorf "%s: src %q resolved to a non-image %s resource at %s" $label $source $kind $position -}} {{- end -}} {{- $src = $resource.RelPermalink -}} {{- $fullSrc = $resource.RelPermalink -}} {{- $mediaType = $resource.MediaType.Type -}} {{- $processable = reflect.IsImageResourceProcessable $resource -}} {{- if reflect.IsImageResourceWithMeta $resource -}} {{- $width = $resource.Width -}} {{- $height = $resource.Height -}} {{- end -}} {{- with $resource.Params -}} {{- if isset . "alt" -}} {{- $metadataAlt := index . "alt" -}} {{- if ne (printf "%T" $metadataAlt) "string" -}} {{- errorf "%s: image resource alt metadata must be a string at %s" $label $position -}} {{- end -}} {{- $resourceAlt = strings.TrimSpace $metadataAlt -}} {{- end -}} {{- with index . "byline" }}{{ $byline = strings.TrimSpace (printf "%v" .) }}{{ end -}} {{- end -}} {{- else -}} {{- $extension := lower (strings.TrimPrefix "." (path.Ext $parsed.Path)) -}} {{- $mediaTypes := dict "avif" "image/avif" "bmp" "image/bmp" "gif" "image/gif" "heic" "image/heic" "heif" "image/heif" "ico" "image/x-icon" "jpeg" "image/jpeg" "jpg" "image/jpeg" "png" "image/png" "svg" "image/svg+xml" "tif" "image/tiff" "tiff" "image/tiff" "webp" "image/webp" -}} {{- with index $mediaTypes $extension }}{{ $mediaType = . }}{{ end -}} {{- end -}} {{- $hasAlt := .hasAlt | default false -}} {{- $alt := $resourceAlt -}} {{- if $hasAlt -}} {{- $altValue := .alt -}} {{- if ne (printf "%T" $altValue) "string" -}} {{- errorf "%s: alt must be a string at %s" $label $position -}} {{- end -}} {{- $alt = strings.TrimSpace $altValue -}} {{- end -}} {{- $decorative := .decorative | default false -}} {{- if ne (printf "%T" $decorative) "bool" -}} {{- errorf "%s: decorative must be boolean at %s" $label $position -}} {{- end -}} {{- if and $decorative $hasAlt $alt -}} {{- errorf "%s: decorative images must not define alt text at %s" $label $position -}} {{- end -}} {{- if $decorative }}{{ $alt = "" }}{{ end -}} {{- if and (.requireAlt | default false) (not $decorative) (not $alt) -}} {{- errorf "%s requires meaningful alt text or decorative=true at %s" $label $position -}} {{- end -}} {{- return (dict "src" $src "fullSrc" $fullSrc "markdownSrc" (partial "content/markdown-url.html" $src) "source" $source "kind" $kind "mediaType" $mediaType "width" $width "height" $height "processable" $processable "processed" false "resource" $resource "renderedResource" $resource "alt" $alt "decorative" $decorative "caption" (.caption | default "") "byline" $byline ) -}}
