{{- /* Context-aware inline Markdown escaping for attribution and metadata text (for example, a quoted social post's author or date line). Unlike content/markdown-escape.html, it preserves ordinary punctuation such as dates, colons, dots, and leading @ signs while neutralizing Markdown syntax, HTML tags/entities, GFM autolinks, and embedded line breaks. */ -}} {{- $text := printf "%v" . -}} {{- $text = strings.TrimSpace (replaceRE `[\r\n\t]+` " " $text) -}} {{- $chars := slice "\\" "*" "_" "~" "[" "]" "<" ">" "|" -}} {{- $chars = $chars | append (printf "%c" 96) -}} {{- range $chars -}} {{- $text = replace $text . (printf "\\%s" .) -}} {{- end -}} {{- $text = replaceRE `(&(?:#[0-9]+|#[xX][0-9a-fA-F]+|[A-Za-z][A-Za-z0-9]+);)` `\${1}` $text -}} {{- $text = replace $text "://" `\://` -}} {{- $text = replaceRE `(?i)(^|[^[:alnum:]_])(www)\.` `${1}${2}\.` $text -}} {{- $text = replaceRE `([A-Za-z0-9.+_-])@` `${1}\@` $text -}} {{- return $text -}}
