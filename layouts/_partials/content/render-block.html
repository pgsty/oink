{{- /* Render a Markdown body (RenderString, block display), then namespace its generated code IDs without sharing mutable state with sibling hooks. Input: page, scope (short token), content (Markdown), origin (optional "shortcode … at file:line" for diagnostics). Returns HTML. */ -}} {{- $page := .page -}} {{- $scope := .scope | default "" -}} {{- $content := .content -}} {{- $origin := .origin | default (printf "page %s" $page.Path) -}} {{- /* Hugo renders a shortcode body in its own Goldmark document, so footnotes cannot work here: a definition on the page around the body is out of reach (Goldmark emits the literal `[^label]`), and a definition inside the body produces a second footnote list whose `fn:N` ids collide with the page's own. Fail instead of publishing either. A reference the page never defines is literal text at page level too, so it passes; code is stripped first, because `[^0-9]` in a listing is a character class. */ -}} {{- $probe := replaceRE "(?s)```.*?```" " " $content -}} {{- $probe = replaceRE "(?s)~~~.*?~~~" " " $probe -}} {{- $probe = replaceRE "`[^`\n]*`" " " $probe -}} {{- range $reference := findRESubmatch `\[\^([^\]\s]+)\]` $probe -}} {{- $label := index $reference 1 -}} {{- $definition := printf "[^%s]:" $label -}} {{- if or (strings.Contains $page.RawContent $definition) (strings.Contains $content $definition) -}} {{- warnf "%s: footnote reference [^%s] cannot be used inside a shortcode body — Hugo renders the body as its own document, so the reference prints literally when the definition sits on the page, and builds a second, colliding footnote list when it sits in the body. Author the block in its native form (a Markdown table, image, or fence followed by {num=\"…\" caption=\"…\"}) and keep footnotes in page Markdown." $origin $label -}} {{- end -}} {{- end -}} {{- $html := $page.RenderString (dict "display" "block") $content -}} {{- if $scope -}} {{- $prefix := printf "td-code-%.8s-" (md5 $page.RelPermalink) -}} {{- range findRESubmatch `id="(td-code-[^"]+)" data-td-code\s+data-td-code-auto-id(?:\s|>)` $html -}} {{- $old := index . 1 -}} {{- if strings.HasPrefix $old $prefix -}} {{- $new := printf "%s%s-%s" $prefix ($scope | anchorize) (strings.TrimPrefix $prefix $old) -}} {{- range (slice "" "-viewport" "-title") -}} {{- $html = replace $html (printf `="%s%s"` $old .) (printf `="%s%s"` $new .) -}} {{- end -}} {{- range findRESubmatch (printf `id="%s-([0-9]+)"` $old) $html -}} {{- $line := index . 1 -}} {{- range $attribute := slice `id="` `href="#` -}} {{- $html = replace $html (printf `%s%s-%s"` $attribute $old $line) (printf `%s%s-%s"` $attribute $new $line) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $html = $html | safeHTML -}} {{- end -}} {{- return $html -}}
