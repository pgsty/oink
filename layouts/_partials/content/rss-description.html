{{- /* Render an RSS-safe summary without crossing a manual summary divider. */ -}} {{- $page := . -}} {{- /* The RSS summary and page feed share this Store. Keep it in RSS state instead of restoring stale HTML state after the summary returns. */ -}} {{- $page.Store.Set "tdOutputFormat" "rss" -}} {{- $markup := $page.Markup "td-rss-summary" -}} {{- $render := $markup.Render -}} {{- $summary := $render.Summary -}} {{- $description := printf "%s" $summary.Text -}} {{- $description = partial "content/static-image-output.html" $description -}} {{- if eq $page.Type "expo" }}{{ $description = printf "%s%s" $description (partial "expo/static.html" $page) }}{{ end -}} {{- return $description -}}
