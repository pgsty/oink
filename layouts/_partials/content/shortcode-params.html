{{/* Reject unknown and mis-shaped shortcode parameters. A typo like `{{< fig num="1-1" captoin="…" >}}` would otherwise build cleanly and silently drop the caption. This warns instead, naming the parameter and the position, and the shortcode renders without it — one mistyped attribute costs that attribute, not every page of the site. --panicOnWarning is what keeps it fatal on publish. Every shortcode calls this, so the check cannot be forgotten for a new one. Parameters: shortcode the calling shortcode's context (`.`) allowed slice of accepted named parameters (default: none) attributes true when the shortcode also mirrors the block-attribute policy (class plus data-* / aria-* passthrough) and validates those values itself — see content/attributes.html form "named" named parameters only (default) "none" the shortcode takes no parameters "positional" positional parameters only "any" a positional shorthand is allowed; named parameters are still checked against `allowed` Returns "" so it composes as {{ partial … }} at the top of a shortcode. */ -}} {{- $sc := .shortcode -}} {{- $allowed := .allowed | default (slice) -}} {{- $form := .form | default "named" -}} {{- $attributes := .attributes | default false -}} {{- $count := 0 -}} {{- with $sc.Params }}{{ $count = len . }}{{ end -}} {{- if eq $form "none" -}} {{- if gt $count 0 -}} {{- warnf "shortcode %q accepts no parameters at %s; ignoring them" $sc.Name $sc.Position -}} {{- end -}} {{- else if eq $form "positional" -}} {{- if $sc.IsNamedParams -}} {{- warnf "shortcode %q accepts positional parameters only at %s; ignoring the named ones" $sc.Name $sc.Position -}} {{- end -}} {{- else -}} {{- if and (gt $count 0) (not $sc.IsNamedParams) (ne $form "any") -}} {{- warnf "shortcode %q accepts named parameters only at %s; ignoring the positional ones" $sc.Name $sc.Position -}} {{- end -}} {{/* A positional shorthand under "any" has integer keys, not names. */ -}} {{- if $sc.IsNamedParams -}} {{- range $name, $_ := $sc.Params -}} {{- $lower := lower $name -}} {{- $passthrough := and $attributes (or (eq $lower "class") (hasPrefix $lower "data-") (hasPrefix $lower "aria-")) -}} {{- if not (or (in $allowed $name) $passthrough) -}} {{- warnf "shortcode %q: unsupported parameter %q at %s (allowed: %s); ignoring it" $sc.Name $name $sc.Position (delimit $allowed ", ") -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return "" -}}
