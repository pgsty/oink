{{- /* Wrapper for one block that carries the `tab` attribute (adjacent-block tabs, plan/design/components.md v5 §3.2). Server output is a titled block; the tabs runtime groups adjacent wrappers of the same kind into one tablist. Input dict: page, position, tab (label), group, value, kind (code|table), content (HTML). Rules: group ^[a-z][a-z0-9_-]*$ (optional, first block of the run); value ^[a-z0-9][a-z0-9_-]*$ (required when the run has a group, forbidden otherwise — validated at runtime across the run and here per block). */ -}} {{- $page := .page -}} {{- $format := lower ($page.Store.Get "tdOutputFormat" | default "html") -}} {{- $tab := strings.TrimSpace (.tab | default "") -}} {{- if not $tab }}{{ warnf "tab: label must not be empty at %s; using the block kind as its label" .position }}{{ $tab = (.kind | default "tab") }}{{ end -}} {{- $group := strings.TrimSpace (.group | default "") -}} {{- $value := strings.TrimSpace (.value | default "") -}} {{- if and $group (not (findRE `^[a-z][a-z0-9_-]*$` $group)) }}{{ warnf "tab: group must match ^[a-z][a-z0-9_-]*$; got %q at %s; dropping the group" $group .position }}{{ $group = "" }}{{ end -}} {{- if and $value (not (findRE `^[a-z0-9][a-z0-9_-]*$` $value)) }}{{ warnf "tab: value must match ^[a-z0-9][a-z0-9_-]*$; got %q at %s; dropping the value" $value .position }}{{ $value = "" }}{{ end -}} {{- if and $group (not $value) }}{{ warnf "tab: value is required when group is declared at %s; the tab will not sync" .position }}{{ $group = "" }}{{ end -}} {{- if eq $format "html" }}{{ $page.Store.Set "hasTabs" true }}{{ end -}}
{{ $tab }}
{{ .content }}
