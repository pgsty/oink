{{- $name := .name | default "content component" -}} {{- $label := .label | default (printf "shortcode %q" $name) -}} {{- $parameter := .parameter | default "URL" -}} {{- $position := .position | default "unknown position" -}} {{- $kind := .kind | default "link" -}} {{- $value := .value -}} {{- $ok := true -}} {{- $raw := "" -}} {{- if ne (printf "%T" $value) "string" -}} {{- warnf "%s: %s must be a string at %s; dropping it" $label $parameter $position -}} {{- $ok = false -}} {{- else -}} {{- $raw = printf "%s" $value -}} {{- if not $raw -}} {{- warnf "%s: %s must not be empty at %s; dropping it" $label $parameter $position -}} {{- $ok = false -}} {{- else if findRE `[\x00-\x20\x7f]` $raw -}} {{- warnf "%s: %s must not contain whitespace or control characters at %s; dropping it" $label $parameter $position -}} {{- $ok = false -}} {{- else if hasPrefix $raw "//" -}} {{- warnf "%s: %s must not use a protocol-relative URL at %s; dropping it" $label $parameter $position -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- $scheme := "" -}} {{- $urlPath := "" -}} {{- $parsed := false -}} {{- if $ok -}} {{- $parsedResult := try (urls.Parse $raw) -}} {{- if $parsedResult.Err -}} {{- warnf "%s: invalid %s %q at %s: %s; dropping it" $label $parameter $raw $position $parsedResult.Err -}} {{- $ok = false -}} {{- else -}} {{- $parsed = $parsedResult.Value -}} {{- $scheme = lower $parsed.Scheme -}} {{- $urlPath = $parsed.Path -}} {{- $allowed := cond (in (slice "image" "web") $kind) (slice "http" "https") (slice "http" "https" "mailto") -}} {{- if and $scheme (not (in $allowed $scheme)) -}} {{- warnf "%s: unsupported %s scheme %q at %s; dropping it" $label $parameter $scheme $position -}} {{- $ok = false -}} {{- else if and (in (slice "http" "https") $scheme) (not $parsed.Host) -}} {{- warnf "%s: %s %q requires a host at %s; dropping it" $label $parameter $raw $position -}} {{- $ok = false -}} {{- else if and (eq $scheme "mailto") (not (or $parsed.Opaque $parsed.Path)) -}} {{- warnf "%s: %s %q requires an address at %s; dropping it" $label $parameter $raw $position -}} {{- $ok = false -}} {{- else if and (eq $kind "image") (not $scheme) (not $parsed.Path) -}} {{- warnf "%s: %s %q requires an image path at %s; dropping it" $label $parameter $raw $position -}} {{- $ok = false -}} {{- end -}} {{- end -}} {{- end -}} {{- $href := "" -}} {{- if $ok -}} {{- $href = $raw -}} {{- if not $scheme -}} {{- if not (or (hasPrefix $raw "#") (hasPrefix $raw "?")) -}} {{- $href = strings.TrimPrefix "/" $raw | relURL -}} {{- end -}} {{- end -}} {{- end -}} {{- $markdown := replace $href `\` `\\` -}} {{- $markdown = replace $markdown "(" `\(` -}} {{- $markdown = replace $markdown ")" `\)` -}} {{- return (dict "href" $href "markdown" $markdown "external" (and $ok (ne $scheme "")) "valid" $ok "raw" (cond $ok $raw "") "scheme" (cond $ok $scheme "") "path" (cond $ok $urlPath "") ) -}}
