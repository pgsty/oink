{{/* Resolve the site's dark-mode policy. `params.ui.dark_mode` is a map with two booleans: enable the theme emits the dark palette and honours a stored choice show_menu the navbar and sidebar expose a System / Light / Dark control `show_menu` implies `enable` unless `enable` is explicitly false, so the common case is one line: ui: dark_mode: show_menu: true Returns a dict with both booleans resolved. */ -}} {{- $config := dict "enable" false "showMenu" false -}} {{- $raw := site.Params.ui.dark_mode -}} {{- if ne $raw nil -}} {{- if eq (printf "%T" $raw) "bool" -}} {{/* Shorthand: `dark_mode: true` turns on the palette and the menu. */ -}} {{- $config = dict "enable" $raw "showMenu" $raw -}} {{- else if reflect.IsMap $raw -}} {{- $enable := index $raw "enable" -}} {{- $showMenu := index $raw "show_menu" -}} {{- if and (ne $enable nil) (ne (printf "%T" $enable) "bool") -}} {{- warnf "params.ui.dark_mode.enable must be a boolean; got %v; ignoring it" $enable -}} {{- $enable = nil -}} {{- end -}} {{- if and (ne $showMenu nil) (ne (printf "%T" $showMenu) "bool") -}} {{- warnf "params.ui.dark_mode.show_menu must be a boolean; got %v; using false" $showMenu -}} {{- $showMenu = false -}} {{- end -}} {{- $showMenu = eq $showMenu true -}} {{- if eq $enable nil }}{{ $enable = $showMenu }}{{ end -}} {{- $config = dict "enable" $enable "showMenu" (and $showMenu $enable) -}} {{- else -}} {{- warnf "params.ui.dark_mode must be a boolean or a map with enable and show_menu; got %v; using false" $raw -}} {{- end -}} {{- end -}} {{- return $config -}}
