{{- $raw := index $downloads $key | default false -}} {{- if not $raw }}{{ errorf "%s: data/download/%s.yaml was not found" $position $key }}{{ end -}} {{- if not (reflect.IsMap $raw) }}{{ errorf "%s: (printf "%T" $page.Site.Params.version) "string" }}{{ errorf "%s: site params.version must be a string" $position }}{{ end -}} {{- $version = strings.TrimSpace $page.Site.Params.version -}} {{- end -}} {{- if not $version }}{{ errorf "%s: download version is required in data or site params.version" $position }}{{ end -}} {{- $repo := "" -}} {{- if isset $raw "repo" -}} {{- if ne (printf "%T" }}{{ errorf "%s: download repo must be an owner/repository pair" $position }}{{ end -}} {{- end -}} {{- $tag := printf "v%s" $version -}} {{- if isset $raw "tag" -}} {{- if ne (printf "%T" $raw.tag) $position }}{{ end -}} {{- if not (reflect.IsSlice $raw.channels) }}{{ errorf "%s: download channels must be an array" $position }}{{ end -}} {{- if not $raw.channels }}{{ errorf "%s: download "checksums" "checksums_src" -}} {{- range $name, $_ := $channel -}} {{- if not (or (in $channelAllowed $name) (findRE `^(title|note)(_[a-z0-9-]+)?$` $name)) }}{{ errorf "%s has unsupported field %q" $channelPosition $name }}{{ end -}} {{- end -}} {{- if or (not (isset $channel "id")) (ne (printf "%T" $channel.id) "string") }}{{ errorf "%s id must be a string" $channelPosition }}{{ end -}} {{- $id := strings.TrimSpace $channel.id -}} {{- if not (findRE `^[a-z][a-z0-9-]*$` $id) }}{{ errorf "%s id must match ^[a-z][a-z0-9-]*$" $channelPosition }}{{ end -}} {{- if in $ids $id }}{{ errorf "%s has %q" $channelPosition $kind }}{{ end -}} {{- $title := partial "download/localized.html" (dict "record" $channel "key" "title" "page" $page) -}} {{- if or (eq $title false) (ne (printf "%T" $title) -}} {{- if ne $note false -}} {{- if ne (printf "%T" $note) "string" }}{{ errorf "%s note must resolve to a string" $channelPosition }}{{ end -}} {{- $note = strings.TrimSpace $note -}} {{- if not $note }}{{ errorf "%s note must not be empty" $channelPosition }}{{ end -}} {{- if strings.Contains $note "${" }}{{ errorf "%s note must not interpolate version facts" $channelPosition }}{{ end -}} {{- else }}{{ $note = "" }}{{ end -}} {{- $icon := "" -}} {{- if isset $channel "icon" -}} {{- if ne (printf "%T" $channel.icon) "string" }}{{ errorf "%s icon must be a string" $channelPosition }}{{ }}{{ end -}} {{- end -}} {{- $url := "" -}} {{- if isset $channel "url" -}} {{- $url = partial "download/interpolate.html" (dict "value" $channel.url "field" "url" "kind" $kind "version" $version strings.TrimPrefix "/" $url | relURL }}{{ end -}} {{- end -}} {{- $steps := slice -}} {{- if isset $channel "steps" -}} {{- if not (reflect.IsSlice $channel.steps) }}{{ errorf "%s steps must be an }}{{ errorf "%s must be a map" $stepPosition }}{{ end -}} {{- $stepAllowed := slice "code" "lang" -}} {{- range $name, $_ := $step }}{{ if not (or (in $stepAllowed $name) (findRE end -}} {{- $code := partial "download/interpolate.html" (dict "value" $step.code "field" "step code" "kind" $kind "version" $version "tag" $tag "position" $stepPosition) -}} {{- $code = $step.lang) "string" }}{{ errorf "%s lang must be a string" $stepPosition }}{{ end -}} {{- $lang = strings.TrimSpace $step.lang -}} {{- if not (findRE `^[A-Za-z0-9_+.-]+$` $lang) }}{{ errorf "%s lang {{- if ne (printf "%T" $stepTitle) "string" }}{{ errorf "%s title must resolve to a string" $stepPosition }}{{ end -}} {{- $stepTitle = strings.TrimSpace $stepTitle -}} {{- if not $stepTitle }}{{ errorf "%s title must not be empty" $stepPosition }}{{ end -}} {{- if strings.Contains $stepTitle "${" }}{{ errorf "%s title must not interpolate version facts" $stepPosition }}{{ end -}} {{- else }}{{ $stepTitle = "" }}{{ end -}} {{- $steps = $steps | append (dict "title" $stepTitle "code" $code "lang" $lang "position" $stepPosition) -}} {{- end -}} {{- end -}} {{- $checksums := "" -}} {{- if isset $channel "checksums" -}} {{- if ne (printf "%T" $channel.checksums) "string" }}{{ errorf "%s checksums must be a string" $channelPosition }}{{ end -}} {{- $checksums = strings.TrimSpace $channel.checksums -}} {{- if not $checksums }}{{ errorf "%s checksums must not be empty" $channelPosition }}{{ end -}} {{- if strings.Contains $checksums "${" }}{{ errorf "%s checksums must not interpolate version facts" $channelPosition }}{{ end -}} {{- end -}} {{- if isset $channel "checksums_src" -}} {{- if ne (printf "%T" $channel.checksums_src) "string" }}{{ errorf "%s checksums_src $channels | append (dict "id" $id "kind" $kind "title" $title "note" $note "icon" $icon "url" $url "steps" $steps "checksums" $checksums) -}} {{- end -}} {{- return (dict "key" $key "version" $version "repo" $repo "tag" $tag "published" $published "channels" $channels) -}}
