{{- /* One top-level Hugo Menu item. Leaves keep the historical link path; parents add a separate disclosure and one dropdown/accordion panel. */ -}} {{- $page := .page -}} {{- $entry := .entry -}} {{- $mode := .mode -}} {{- $index := .index -}} {{- $hasChildren := $entry.HasChildren -}} {{- $taxonomyPage := false -}} {{- with $entry.Page -}} {{- if eq .Kind "taxonomy" }}{{ $taxonomyPage = . }}{{ end -}} {{- end -}} {{- /* URL-based menus predate pageRef and are still common in consumers. Resolve their local target so a /tags/ entry gets the same taxonomy panel without forcing a configuration migration. */ -}} {{- if and (not $taxonomyPage) $entry.URL -}} {{- $target := urls.Parse $entry.URL -}} {{- if not $target.Host -}} {{- $targetPage := $page.Site.GetPage $target.Path -}} {{- if and (not $targetPage) $page.Site.LanguagePrefix (hasPrefix $target.Path $page.Site.LanguagePrefix) -}} {{- $targetPage = $page.Site.GetPage (strings.TrimPrefix $page.Site.LanguagePrefix $target.Path) -}} {{- end -}} {{- with $targetPage -}} {{- if eq .Kind "taxonomy" }}{{ $taxonomyPage = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- $hasPanel := or $hasChildren $taxonomyPage -}} {{- $key := $entry.Identifier | default $entry.Name | urlize -}} {{- $panelID := printf "td-navbar-%s-%s-%d" $mode $key $index -}} {{- if not $hasPanel -}} {{- if eq $mode "desktop" -}} {{ partial "navbar-entry-link.html" (dict "page" $page "entry" $entry "class" "nav-link") }} {{- else -}} {{ partial "navbar-entry-link.html" (dict "page" $page "entry" $entry "class" "td-site-nav__menu-link") }} {{- end -}} {{- else if eq $mode "desktop" -}} {{- $columns := 1 -}} {{- with $entry.Params.columns -}} {{- if not (findRE `^[1-4]$` (printf "%v" .)) -}} {{- warnf "navbar menu %q: columns must be an integer from 1 to 4; using 1" $entry.Name -}} {{- end -}} {{- $columns = int . -}} {{- if or (lt $columns 1) (gt $columns 4) -}} {{- warnf "navbar menu %q: columns must be from 1 to 4; got %d; using 1" $entry.Name $columns -}}{{- $columns = 1 -}} {{- end -}} {{- end -}} {{- /* No disclosure caret on desktop: the parent label navigates on click and the panel opens on hover or keyboard focus (see navbar-menu.js). Touch users navigate directly; the target page carries its own navigation. */ -}}
{{ partial "navbar-entry-link.html" (dict "page" $page "entry" $entry "class" "nav-link td-nav-menu__parent-link" "controls" $panelID) }}
{{- if $taxonomyPage -}} {{ partialCached "navbar-taxonomy-tags.html" (dict "page" $page "taxonomyPage" $taxonomyPage) $page.Site.Language.Lang $taxonomyPage.RelPermalink }} {{- else -}} {{ partial "navbar-group-items.html" (dict "page" $page "items" $entry.Children "mode" "desktop" "top" $entry "depth" 1) }} {{- end }}
{{- else -}} {{- /* Compact menus navigate to the parent landing; that page exposes the hierarchy in its own shell. Without this branch taxonomy parents would disappear whenever this partial is reused in mobile mode. */ -}} {{ partial "navbar-entry-link.html" (dict "page" $page "entry" $entry "class" "td-site-nav__menu-link") }} {{- end -}}
