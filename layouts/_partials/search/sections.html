{{- /* Read the anchors of a page from its rendered HTML: every heading, and every element that names itself a search entry with `data-td-search-entry="Title"`, such as a field. IDs are the rendered ones, including render hooks and shortcodes, never guessed slugs. Offsets are UTF-8 bytes in the emitted body, not JavaScript character positions. Body text is stored once per page, divided among the anchors in order. */ -}} {{- $content := .page.Content | string -}} {{- $content = replaceRE `(?is)<(script|style)\b[^>]*>.*?` "" $content -}} {{- $attributes := `(?:[^>"']|"[^"]*"|'[^']*')*` -}} {{- $headingPattern := printf `(?is)().*?` $attributes -}} {{- $value := `(?:"([^"]*)"|'([^']*)'|([^\s>]+))` -}} {{- $entry := printf `(?is)<[a-z][a-z0-9:._-]*%s\sdata-td-search-entry\s*=\s*%s%s>` $attributes $value $attributes -}} {{- $attribute := printf `(?is)\s+([^\s=/>]+)(?:\s*=\s*%s)?` $value -}} {{- $marker := "\x1eOINK-SEARCH\x1e" -}} {{- $marked := replaceRE $headingPattern (printf "%s${0}" $marker) $content -}} {{- $sections := slice -}} {{- $headings := slice -}} {{- $body := "" -}} {{- range $block := split $marked $marker -}} {{- $tag := "" -}} {{- $headingTitle := "" -}} {{- range findRESubmatch (printf "^%s" $headingPattern) $block 1 -}} {{- $tag = index . 1 -}} {{- $headingTitle = index . 0 | plainify | htmlUnescape -}} {{- $block = strings.TrimPrefix $tag $block -}} {{- end -}} {{- range $index, $part := split (replaceRE $entry (printf "%s${0}" $marker) $block) $marker -}} {{- $isHeading := and (eq $index 0) $tag -}} {{- if $index -}} {{- range findRE (printf "^%s" $entry) $part 1 }}{{ $tag = . }}{{ end -}} {{- end -}} {{- $text := "" -}} {{- if eq $.scope "content" -}} {{- $text = replaceRE `(?i)|` " " $part | plainify | htmlUnescape | replaceRE `\s+` " " -}} {{- if hasPrefix $text " " -}} {{- if and $body (not (hasSuffix $body " ")) }}{{ $body = print $body " " }}{{ end -}} {{- $text = strings.TrimPrefix " " $text -}} {{- end -}} {{- end -}} {{- if $tag -}} {{- $id := "" -}} {{- $title := "" -}} {{- range findRESubmatch $attribute $tag -}} {{- $name := lower (index . 1) -}} {{- if eq $name "id" }}{{ $id = or (index . 2) (index . 3) (index . 4) | htmlUnescape -}} {{- else if eq $name "data-td-search-entry" }}{{ $title = or (index . 2) (index . 3) (index . 4) | htmlUnescape -}} {{- end -}} {{- end -}} {{- if $isHeading }}{{ $title = $headingTitle }}{{ end -}} {{- $title = replaceRE `\s+` " " $title | strings.TrimSpace -}} {{- if and $id (or $isHeading $title) -}} {{- $section := dict "id" $id "title" $title -}} {{- if eq $.scope "content" }}{{ $section = merge $section (dict "start" (len $body)) }}{{ end -}} {{- $sections = $sections | append $section -}} {{- if $isHeading }}{{ $headings = $headings | append $title }}{{ end -}} {{- else if not $isHeading -}} {{- warnf "%s: a data-td-search-entry element needs an id and a non-empty title; skipping it" ($.page.Path | default $.page.RelPermalink) -}} {{- end -}} {{- end -}} {{- if $text }}{{ $body = print $body $text }}{{ end -}} {{- end -}} {{- end -}} {{- return (dict "sections" $sections "headings" $headings "body" $body) -}}
