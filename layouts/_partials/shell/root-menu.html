Entries are the section itself (the default, e.g. Docs) plus each opted-in descendant; sibling top-level sections are the top navbar's job, not this menu's. params.ui.sidebar_root_menu gates the dropdown. Accepts a page, or a dict {page, fallback}. With `fallback: true` (the navbar-mode root row) a page without any switchable descendant still gets a static link to its section, so the row always names where you are and clicking it returns to the section landing. */ -}} {{- $p := . -}} {{- $fallback := false -}} {{- if reflect.IsMap . -}} {{- $p = .page -}} {{- $fallback = .fallback | default false -}} {{- end -}} {{- $top := $p.FirstSection -}} {{- if in (slice "taxonomy" "term") $p.Kind -}} {{- /* Taxonomy pages have no content ancestry; a term adopts its members' shared top section (mirroring the sidebar tree), and mixed or empty cases render nothing. */ -}} {{- $top = "" -}} {{- if eq $p.Kind "term" -}} {{- $tops := slice -}} {{- range $p.Pages }}{{ with .FirstSection }}{{ $tops = $tops | append . }}{{ end }}{{ end -}} {{- $tops = $tops | uniq -}} {{- if eq (len $tops) 1 }}{{ $top = index $tops 0 }}{{ end -}} {{- end -}} {{- end -}} {{- $rendered := false -}} {{- if and $p.Site.Params.ui.sidebar_root_menu $top (not $top.IsHome) -}} {{- $subRoots := slice -}} {{- range where $p.Site.Pages "Params.sidebar_root_for" "self" -}} {{- if and (ne . $top) (.IsDescendant $top) }}{{ $subRoots = $subRoots | append . }}{{ end -}} {{- end -}} {{- if gt (len $subRoots) 0 -}} {{- $rendered = true -}} {{- $entries := slice $top -}} {{- range $subRoots }}{{ $entries = $entries | append . }}{{ end -}} {{- $current := $top -}} {{- range $subRoots -}} {{- if or (eq $p .) ($p.IsDescendant .) }}{{ $current = . }}{{ end -}} {{- end -}}
