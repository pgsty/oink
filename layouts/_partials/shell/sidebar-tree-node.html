{{/* Walk one content-tree node; sidebar-node.html owns shared row markup. cSpell:ignore relref */ -}} {{- $s := .section -}} {{- $p := .page -}} {{- $shouldDelayActive := .shouldDelayActive -}} {{- $sidebarMenuTruncate := .sidebarMenuTruncate -}} {{- $treeRoot := eq .ulNr 0 -}} {{- $ulNr := .ulNr -}} {{- $ulShow := .ulShow -}} {{- $iconPolicy := .iconPolicy -}} {{- $compact := .compact -}} {{- $foldableEnabled := .foldable -}} {{- $overflow := .overflow -}} {{- $active := and (not $shouldDelayActive) (eq $s $p) -}} {{- $activePath := and (not $shouldDelayActive) (or (eq $p $s) ($p.IsDescendant $s)) -}} {{/* Blog sections start expanded; sidebar_expanded: false collapses one. isset keeps an explicit false distinct from the type default. */ -}} {{- $forceExpanded := eq $s.Type "blog" -}} {{- if isset $s.Params "sidebar_expanded" -}} {{- $forceExpanded = $s.Params.sidebar_expanded -}} {{- end -}} {{- $shouldExpand := or $activePath $forceExpanded -}} {{- $show := cond (or (lt $ulNr $ulShow) $activePath (and (not $shouldDelayActive) (eq $s.Parent $p.Parent)) (and (not $shouldDelayActive) (eq $s.Parent $p)) (not $compact) (and (not $shouldDelayActive) ($p.IsDescendant $s.Parent)) ) true false -}} {{- $children := union $s.Pages $s.Sections -}} {{- $orderedChildren := $children.ByWeight -}} {{- if eq $s.Type "blog" -}} {{- $orderedChildren = partial "shell/blog-pages.html" $children -}} {{- end -}} {{- $pages_tmp := where $orderedChildren ".Params.toc_hide" "!=" true -}} {{- $pages := $pages_tmp | first $sidebarMenuTruncate -}} {{- if gt (len $pages_tmp) $sidebarMenuTruncate -}} {{- $truncatedEntryCount := sub (len $pages_tmp) $sidebarMenuTruncate -}} {{- warnf "sidebar: %d entries were truncated in section %[4]s; raise params.ui.sidebar_menu_truncate to at least %[2]d (from %[3]d)" $truncatedEntryCount (len $pages_tmp) $sidebarMenuTruncate $s.Path -}} {{- end -}} {{- $withChild := gt (len $pages) 0 -}} {{- $foldable := and $foldableEnabled (ge $ulNr 1) $withChild -}} {{- $childMarkup := "" -}} {{- if $withChild -}} {{- $childUlNr := add $ulNr 1 -}} {{- $rows := "" -}} {{- range $pages -}} {{- if (not (and (eq $s $p.Site.Home) (eq .Params.toc_root true))) -}} {{- $child := dict "page" $p "section" . "shouldDelayActive" $shouldDelayActive "sidebarMenuTruncate" $sidebarMenuTruncate "ulNr" $childUlNr "ulShow" $ulShow "iconPolicy" $iconPolicy "compact" $compact "foldable" $foldableEnabled "overflow" $overflow -}} {{- if or $shouldDelayActive (eq $p .) ($p.IsDescendant .) -}} {{- $rows = printf "%s%s" $rows (partial "shell/sidebar-tree-node.html" $child) -}} {{- else -}} {{- /* A branch the current page is not in renders the same on every page that sees it from the same depth with the same settings: no row in it is active, on the active path, or the page's own Book headings. The one thing the page still decides is whether the branch's own row is hidden, which reads the row's parent alone, so that bit joins the key. Below the row nothing reads the page. A site under sidebar_cache_limit therefore walks each branch once instead of once per page, and its output does not change. */ -}} {{- $hidden := not (or (lt $childUlNr $ulShow) (eq .Parent $p.Parent) (eq .Parent $p) (not $compact) ($p.IsDescendant .Parent) $foldableEnabled) -}} {{- $rows = printf "%s%s" $rows (partialCached "shell/sidebar-tree-node.html" $child $p.Site.BaseURL $p.Language.Lang .Path $childUlNr $ulShow $hidden $iconPolicy $compact $foldableEnabled $overflow $sidebarMenuTruncate) -}} {{- end -}} {{- end -}} {{- end -}} {{- $childMarkup = printf "
    %s
" $childUlNr $rows | safeHTML -}} {{- end -}} {{/* In a foldable tree the parent disclosure owns child visibility. Leaves have no toggle of their own, but must remain visible inside an open branch. */ -}} {{- partial "shell/sidebar-node.html" (dict "page" $p "section" $s "active" $active "activePath" $activePath "treeRoot" $treeRoot "iconRoot" (le .ulNr 1) "withChild" $withChild "foldable" $foldable "shouldExpand" $shouldExpand "hidden" (not (or $show $foldableEnabled)) "iconPolicy" $iconPolicy "overflow" $overflow "children" $childMarkup ) -}}
