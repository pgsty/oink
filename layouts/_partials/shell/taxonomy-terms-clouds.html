{{- /* Scope every taxonomy cloud to the current top-level content root. A docs page therefore counts only docs descendants; blog pages count only blog descendants. Taxonomy term pages inherit their configured type. */ -}} {{- $root := .FirstSection -}} {{- with .Site.GetPage (printf "/%s" .Type) -}} {{- if .IsSection }}{{ $root = . }}{{ end -}} {{- end -}} {{- /* Term pages have no content ancestry; adopt their members' shared top section (mirroring the sidebar tree) so a docs term shows docs-scoped clouds, not the site-wide mix. */ -}} {{- if eq .Kind "term" -}} {{- $tops := slice -}} {{- range .Pages }}{{ with .FirstSection }}{{ $tops = $tops | append . }}{{ end }}{{ end -}} {{- $tops = $tops | uniq -}} {{- if eq (len $tops) 1 }}{{ $root = index $tops 0 }}{{ end -}} {{- end -}} {{- $global := or (not $root) (eq $root .Site.Home) -}} {{- range $plural, $taxonomy := .Site.Taxonomies -}} {{- $items := slice -}} {{- range $taxonomy.ByCount -}} {{- $count := 0 -}} {{- range .Pages -}} {{- if or $global (eq . $root) (.IsDescendant $root) -}} {{- $count = add $count 1 -}} {{- end -}} {{- end -}} {{- if gt $count 0 -}} {{- $items = $items | append (dict "count" $count "page" .Page) -}} {{- end -}} {{- end -}} {{- if gt (len $items) 0 -}} {{- $title := partial "taxonomy-label.html" (dict "page" $ "key" $plural) -}} {{- $id := printf "td-shell-taxonomy-%s-%s" $plural ($root.RelPermalink | default "all" | anchorize) -}} {{- /* Every aside group carries an icon. Sites may override per taxonomy via params.ui.taxonomy_icons.; unknown taxonomies get a generic glyph. */ -}} {{- $icon := "" -}} {{- with $.Site.Params.ui.taxonomy_icons }}{{ with index . $plural }}{{ $icon = . }}{{ end }}{{ end -}} {{- if not $icon -}} {{- $icon = index (dict "categories" "fa-solid fa-folder" "tags" "fa-solid fa-tags") $plural | default "fa-solid fa-shapes" -}} {{- end -}}
