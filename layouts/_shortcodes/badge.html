{{- /* Semantic inline status label with static fallbacks for every output. */ -}} {{- partial "content/shortcode-params.html" (dict "shortcode" . "allowed" (slice "text" "tone" "link")) -}} {{- if not (and .IsNamedParams (isset .Params "text")) -}} {{- errorf "shortcode %q requires parameter text at %s" .Name .Position -}} {{- end -}} {{- $textParam := .Get "text" -}} {{- if ne (printf "%T" $textParam) "string" -}} {{- errorf "shortcode %q: text must be a string at %s" .Name .Position -}} {{- end -}} {{- $badgeText := strings.TrimSpace $textParam -}} {{- if not $badgeText -}} {{- errorf "shortcode %q: text must not be empty at %s" .Name .Position -}} {{- end -}} {{- $tone := "neutral" -}} {{- if and .IsNamedParams (isset .Params "tone") -}} {{- $toneParam := .Get "tone" -}} {{- if ne (printf "%T" $toneParam) "string" -}} {{- errorf "shortcode %q: tone must be a string at %s" .Name .Position -}} {{- end -}} {{- $tone = strings.TrimSpace $toneParam -}} {{- end -}} {{- $tones := slice "neutral" "info" "success" "warning" "danger" -}} {{- if not (in $tones $tone) -}} {{- errorf "shortcode %q: tone must be one of %s; got %q at %s" .Name (delimit $tones " | ") $tone .Position -}} {{- end -}} {{- $link := false -}} {{- if and .IsNamedParams (isset .Params "link") -}} {{- $link = partial "content/url.html" (dict "name" .Name "parameter" "link" "position" .Position "value" (.Get "link") ) -}} {{- end -}} {{- $outputFormat := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $outputFormat "markdown" -}} {{- $markdownText := partial "content/markdown-escape.html" $badgeText -}} {{- if $link -}} {{- printf "[**%s**](%s)" $markdownText $link.markdown | safeHTML -}} {{- else -}} {{- printf "**%s**" $markdownText | safeHTML -}} {{- end -}} {{- else -}} {{- $class := printf "td-badge td-badge--%s" $tone -}} {{- if $link -}} {{ $badgeText }} {{- else -}} {{ $badgeText }} {{- end -}} {{- end -}}
