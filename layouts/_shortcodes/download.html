{{- /* Render a validated data/download/ record; facts are not params. A malformed call warns and renders nothing rather than stopping the build: one mistyped shortcode should cost its own block, not every page of the site. resolve.html degrades the same way for a malformed record. */ -}} {{- partial "content/shortcode-params.html" (dict "shortcode" . "form" "positional") -}} {{- $ok := true -}} {{- if or .IsNamedParams (ne (len .Params) 1) -}} {{- warnf "shortcode %q requires exactly one positional data key at %s; skipping the block" .Name .Position -}} {{- $ok = false -}} {{- end -}} {{- $key := "" -}} {{- if $ok -}} {{- $keyParam := .Get 0 -}} {{- if ne (printf "%T" $keyParam) "string" -}} {{- warnf "shortcode %q data key must be a string at %s; skipping the block" .Name .Position -}} {{- $ok = false -}} {{- else -}} {{- $key = strings.TrimSpace $keyParam -}} {{- end -}} {{- end -}} {{- if $ok -}} {{- $owner := printf "download-%d" .Ordinal -}} {{- $owners := .Page.Store.Get "tdDownloadOwners" | default (dict) -}} {{- if isset $owners $key -}} {{- if ne (index $owners $key) $owner -}} {{- /* Rendering the same record twice would duplicate every element id on the page, so the repeat is dropped rather than emitted. */ -}} {{- warnf "shortcode %q: duplicate data key %q at %s; skipping the repeat" .Name $key .Position -}} {{- $ok = false -}} {{- end -}} {{- else -}} {{- .Page.Store.SetInMap "tdDownloadOwners" $key $owner -}} {{- end -}} {{- if $ok -}} {{- $data := partial "download/resolve.html" (dict "page" .Page "key" $key "position" .Position) -}} {{- partial "download/render.html" (dict "page" .Page "data" $data "owner" $owner) -}} {{- end -}} {{- end -}}
