{{- /* Render ordered field definitions collected by field children (`<` collector: children keep raw Markdown descriptions, the parent renders them with RenderString for HTML / print / RSS and emits the source Markdown for the Markdown output). Shared
renderer: content/fields-list.html (also used by the `{.fields}` table hook). */ -}} {{- $hasParams := false -}} {{- with .Params }}{{ $hasParams = true }}{{ end -}} {{- if and $hasParams (not .IsNamedParams) -}} {{- errorf "shortcode %q accepts named parameters only at %s" .Name .Position -}} {{- end -}} {{- $allowed := slice "label" -}} {{- range $name, $_ := .Params -}} {{- if not (in $allowed $name) -}} {{- errorf "shortcode %q: unsupported parameter %q at %s" $.Name $name $.Position -}} {{- end -}} {{- end -}} {{- $label := "" -}} {{- if and $hasParams (isset .Params "label") -}} {{- $labelParam := .Get "label" -}} {{- if ne (printf "%T" $labelParam) "string" -}} {{- errorf "shortcode %q: label must be a string at %s" .Name .Position -}} {{- end -}} {{- $label = strings.TrimSpace $labelParam -}} {{- if not $label -}} {{- errorf "shortcode %q: label must not be empty at %s" .Name .Position -}} {{- end -}} {{- end -}} {{- $innerOutput := strings.TrimSpace (printf "%s" .Inner) -}} {{- if $innerOutput -}} {{- errorf "shortcode %q accepts field children only at %s" .Name .Position -}} {{- end -}} {{- $entries := .Scratch.Get "fieldEntries" | default (dict) -}} {{- if eq (len $entries) 0 -}} {{- errorf "shortcode %q requires at least one field child at %s" .Name .Position -}} {{- end -}} {{- $outputFormat := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $outputFormat "markdown" -}} {{- $chunks := slice -}} {{- range $entries -}} {{- $fieldName := partial "content/markdown-code-span.html" .name -}} {{- $metadata := slice -}} {{- with .type -}} {{- $metadata = $metadata | append (partial "content/markdown-code-span.html" .) -}} {{- end -}} {{- if .required -}} {{- /* Deliberately untranslated API vocabulary; see the contract. */ -}} {{- $metadata = $metadata | append "required" -}} {{- end -}} {{- if .hasDefault -}} {{- $defaultValue := partial "content/markdown-code-span.html" .default -}} {{- $metadata = $metadata | append (printf "default: %s" $defaultValue) -}} {{- end -}} {{- $heading := printf "- %s" $fieldName -}} {{- if gt (len $metadata) 0 -}} {{- $heading = printf "%s — %s" $heading (delimit $metadata "; ") -}} {{- end -}} {{- $description := partial "content/markdown-indent.html" .description -}} {{- $chunks = $chunks | append (printf "%s\n\n%s" $heading $description) -}} {{- end -}} {{- $markdown := delimit $chunks "\n\n" -}} {{- if $label -}} {{- $markdownLabel := partial "content/markdown-escape.html" $label -}} {{- $markdown = printf "**%s**\n\n%s" $markdownLabel $markdown -}} {{- end -}} {{- $markdown | safeHTML -}} {{- else -}} {{- $labelID := printf "td-fields-%s-%d-label" (md5 .Page.RelPermalink) .Ordinal -}} {{- $items := slice -}} {{- range $entries -}} {{- $meta := slice -}} {{- with .type -}} {{- $meta = $meta | append (dict "label" "" "value" (printf "%s" (htmlEscape .)) "class" "td-field__type") -}} {{- end -}} {{- if .required -}} {{- /* required/default are untranslated API vocabulary; see the contract. */ -}} {{- $meta = $meta | append (dict "label" "required" "value" "" "class" "td-field__required") -}} {{- end -}} {{- if .hasDefault -}} {{- $meta = $meta | append (dict "label" "default: " "value" (printf "%s" (htmlEscape .default)) "class" "td-field__default") -}} {{- end -}} {{- $items = $items | append (dict "name" (htmlEscape .name) "meta" $meta "description" ($.Page.RenderString (dict "display" "block") .description) "descriptionMarkdown" false) -}} {{- end -}} {{- partial "content/fields-list.html" (dict "label" $label "labelID" $labelID "entries" $items) -}} {{- end -}}
