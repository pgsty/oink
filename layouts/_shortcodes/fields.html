{{- /* Render ordered field definitions collected by field children (`<` collector: children keep raw Markdown descriptions, the parent renders them with RenderString for HTML / print / RSS and emits the source Markdown for the Markdown output). Shared
renderer: content/fields-list.html (also used by the `{.fields}` table hook). */ -}} {{- partial "content/shortcode-params.html" (dict "shortcode" . "attributes" true "allowed" (slice "label" "id")) -}} {{- /* Container attributes mirror the `{.fields}` table hook (content/attributes.html): named parameters, plus class token validation and data- / aria- passthrough, so the same block can be styled and targeted whichever form wrote it. */ -}} {{- $allowed := slice "label" "id" -}} {{- $generic := dict -}} {{- $classes := slice -}} {{- range $name, $value := .Params -}} {{- $lower := lower $name -}} {{- if in $allowed $lower -}} {{- else if eq $lower "class" -}} {{- range $token := split (strings.TrimSpace (printf "%v" $value)) " " -}} {{- $token = strings.TrimSpace $token -}} {{- if $token -}} {{- if not (findRE `^[A-Za-z0-9_-]+$` $token) -}} {{- warnf "shortcode %q: class token %q contains unsupported characters at %s; ignoring it" $.Name $token $.Position -}}{{- continue -}} {{- end -}} {{- if not (in $classes $token) }}{{ $classes = $classes | append $token }}{{ end -}} {{- end -}} {{- end -}} {{- else if or (strings.HasPrefix $lower "on") (eq $lower "style") (eq $lower "srcdoc") -}} {{- warnf "shortcode %q: unsafe attribute %q is not supported at %s; ignoring it" $.Name $name $.Position -}} {{- else if or (strings.HasPrefix $lower "data-") (strings.HasPrefix $lower "aria-") -}} {{- if not (findRE `^[a-z][a-z0-9_:.-]*$` $lower) -}} {{- warnf "shortcode %q: invalid attribute name %q at %s; ignoring it" $.Name $name $.Position -}} {{- end -}} {{- $generic = merge $generic (dict $lower (printf "%v" $value)) -}} {{- else -}} {{- warnf "shortcode %q: unsupported parameter %q at %s; ignoring it" $.Name $name $.Position -}} {{- end -}} {{- end -}} {{- $id := "" -}} {{- if and .IsNamedParams (isset .Params "id") -}} {{- $id = strings.TrimSpace (printf "%v" (.Get "id")) -}} {{- if not $id -}} {{- warnf "shortcode %q: id must not be empty at %s; dropping it" .Name .Position -}} {{- end -}} {{- if findRE `[\s"'<>&]` $id -}} {{- warnf "shortcode %q: id %q contains unsupported characters at %s; dropping it" .Name $id .Position -}} {{- end -}} {{- end -}} {{- $label := "" -}} {{- if and .IsNamedParams (isset .Params "label") -}} {{- $labelParam := .Get "label" -}} {{- if ne (printf "%T" $labelParam) "string" -}} {{- warnf "shortcode %q: label must be a string at %s; dropping it" .Name .Position -}} {{- end -}} {{- $label = strings.TrimSpace $labelParam -}} {{- if not $label -}} {{- warnf "shortcode %q: label must not be empty at %s; dropping it" .Name .Position -}} {{- end -}} {{- end -}} {{- $innerOutput := strings.TrimSpace (printf "%s" .Inner) -}} {{- if $innerOutput -}} {{- warnf "shortcode %q accepts field children only at %s; ignoring the stray content" .Name .Position -}} {{- end -}} {{- $entries := .Scratch.Get "fieldEntries" | default (dict) -}} {{- if eq (len $entries) 0 -}} {{- warnf "shortcode %q requires at least one field child at %s; rendering nothing" .Name .Position -}} {{- end -}} {{- $outputFormat := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $outputFormat "markdown" -}} {{- $chunks := slice -}} {{- range $entries -}} {{- $fieldName := partial "content/markdown-code-span.html" .name -}} {{- $metadata := slice -}} {{- with .type -}} {{- $metadata = $metadata | append (partial "content/markdown-code-span.html" .) -}} {{- end -}} {{- if .required -}} {{- /* Deliberately untranslated API vocabulary; see the contract. */ -}} {{- $metadata = $metadata | append "required" -}} {{- end -}} {{- if .hasDefault -}} {{- $defaultValue := partial "content/markdown-code-span.html" .default -}} {{- $metadata = $metadata | append (printf "default: %s" $defaultValue) -}} {{- end -}} {{- $heading := printf "- %s" $fieldName -}} {{- if gt (len $metadata) 0 -}} {{- $heading = printf "%s — %s" $heading (delimit $metadata "; ") -}} {{- end -}} {{- $description := partial "content/markdown-indent.html" .description -}} {{- $chunks = $chunks | append (printf "%s\n\n%s" $heading $description) -}} {{- end -}} {{- $markdown := delimit $chunks "\n\n" -}} {{- if $label -}} {{- $markdownLabel := partial "content/markdown-escape.html" $label -}} {{- $markdown = printf "**%s**\n\n%s" $markdownLabel $markdown -}} {{- end -}} {{- $markdown | safeHTML -}} {{- else -}} {{- $labelID := printf "td-fields-%s-%d-label" (md5 .Page.RelPermalink) .Ordinal -}} {{- $items := slice -}} {{- range $key, $entry := $entries -}} {{- $meta := partial "content/field-meta.html" (dict "type" $entry.type "required" $entry.required "hasDefault" $entry.hasDefault "default" $entry.default ) -}} {{- $anchor := partial "content/field-anchor.html" (dict "page" $.Page "name" $entry.name "identity" (printf "sc%d#%s" $.Ordinal $key)) -}} {{- /* The anchor is also the render scope. A scope derived from the name instead is not unique -- two names that slug alike (`a.b` and `a-b`) produce one scope, and the ids that nested hooks generate inside both descriptions collide, which is a duplicate id in the document, not merely an ugly one. The anchor registry has already resolved exactly that collision to allocate `field-a-b` and `field-a-b-2`, so reusing its answer is the only source of uniqueness worth having here. */ -}} {{- $items = $items | append (dict "name" (htmlEscape $entry.name) "anchor" $anchor "meta" $meta "description" (partial "content/render-block.html" (dict "page" $.Page "scope" $anchor "content" $entry.description "origin" (printf "shortcode %q at %s" $.Name $.Position))) "descriptionMarkdown" false) -}} {{- end -}} {{- partial "content/fields-list.html" (dict "label" $label "labelID" $labelID "entries" $items "format" $outputFormat "id" $id "class" (delimit $classes " ") "generic" $generic ) -}} {{- end -}}
