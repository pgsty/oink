{{- /* Positional-only keyboard sequence with a pure Markdown fallback. */ -}} {{- partial "content/shortcode-params.html" (dict "shortcode" . "form" "positional") -}} {{- if eq (len .Params) 0 -}} {{- errorf "shortcode %q requires at least one key at %s" .Name .Position -}} {{- end -}} {{- $keys := slice -}} {{- range .Params -}} {{- if ne (printf "%T" .) "string" -}} {{- errorf "shortcode %q: every key must be a string at %s" $.Name $.Position -}} {{- end -}} {{- $key := strings.TrimSpace . -}} {{- if not $key -}} {{- errorf "shortcode %q: keys must not be empty at %s" $.Name $.Position -}} {{- end -}} {{- $keys = $keys | append $key -}} {{- end -}} {{- $outputFormat := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if eq $outputFormat "markdown" -}} {{- $escaped := slice -}} {{- range $keys -}} {{- $escaped = $escaped | append (partial "content/markdown-escape.html" .) -}} {{- end -}} {{- delimit $escaped " + " | safeHTML -}} {{- else -}} {{- range $index, $key := $keys -}} {{- if gt $index 0 -}} + {{ T "ui_kbd_with" }} {{- end -}} {{ $key }} {{- end -}} {{- end -}}
