{{- /* One tab inside `{{< tabs >}}`: label (required), value (required when the parent declares group, forbidden otherwise); the body is Markdown. */ -}} {{- with .Parent -}} {{- if ne .Name "tabs" }}{{ errorf "shortcode %q must be enclosed by tabs, not %q, at %s" $.Name .Name $.Position }}{{ end -}} {{- else -}} {{- errorf "shortcode %q must be enclosed by tabs at %s" .Name .Position -}} {{- end -}} {{- if or (not .IsNamedParams) (not (isset .Params "label")) }}{{ errorf "shortcode %q requires named parameter label at %s" .Name .Position }}{{ end -}} {{- range $name, $_ := .Params -}} {{- if not (in (slice "label" "value") $name) }}{{ errorf "shortcode %q: unsupported parameter %q at %s" $.Name $name $.Position }}{{ end -}} {{- end -}} {{- $label := strings.TrimSpace (printf "%v" (.Get "label")) -}} {{- if not $label }}{{ errorf "shortcode %q: label must not be empty at %s" .Name .Position }}{{ end -}} {{- $value := "" -}} {{- if isset .Params "value" -}} {{- $value = strings.TrimSpace (printf "%v" (.Get "value")) -}} {{- if not (findRE `^[a-z0-9][a-z0-9_-]*$` $value) }}{{ errorf "shortcode %q: value must match ^[a-z0-9][a-z0-9_-]*$; got %q at %s" .Name $value .Position }}{{ end -}} {{- end -}} {{- $body := strings.TrimSpace (printf "%s" .InnerDeindent) -}} {{- .Parent.Scratch.SetInMap "tabEntries" (printf "%06d" .Ordinal) (dict "label" $label "value" $value "body" $body "position" (printf "%s" .Position)) -}}
