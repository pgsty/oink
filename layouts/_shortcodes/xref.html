rendering the text without a link" .Name .Position }}{{ end -}} {{- if and (not $anchor) (not $missingTarget) }}{{ $anchor = printf "%s-%s" $kind $num }}{{ end -}} {{- $pageRef := "" -}} {{- $targetPage := .Page -}} {{- if isset $params "page" -}} {{- $pageRef = .Get "page" -}} {{- if or (ne (printf "%T" $pageRef) "string") (not (strings.TrimSpace $pageRef)) }}{{ warnf "shortcode %q: page must be a non-empty string at %s; resolving on this page" .Name .Position }}{{ $pageRef = "" }}{{ end -}} {{- $pageRef = strings.TrimSpace $pageRef -}} {{- if $pageRef -}} {{- $targetPage = .Page.GetPage $pageRef | default (.Page.Site.GetPage $pageRef) -}} {{- if not $targetPage }}{{ warnf "shortcode %q: page %q was not found in language %s at %s; rendering the text without a link" .Name $pageRef .Page.Language.Lang .Position }}{{ $missingTarget = true }}{{ end -}} {{- end -}} {{- end -}} {{- /* Registration is diagnostic only; rendering never depends on target order. */ -}} {{- .Page.Store.SetInMap "tdBookXrefs" (printf "%08d" .Ordinal) (dict "kind" $kind "num" $num "anchor" $anchor "pageRef" $pageRef "targetPage" $targetPage "position" .Position ) -}} {{- $href := printf "#%s" $anchor -}} {{- if and $pageRef $targetPage }}{{ $href = printf "%s#%s" $targetPage.RelPermalink $anchor }}{{ end -}} {{- $inner := strings.TrimSpace (printf "%s" .InnerDeindent) -}} {{- $label := "" -}} {{- if $inner }}{{ $label = .Page.RenderString (dict "display" "inline") $inner }}{{ else if $kind }}{{ $label = partial "book/label.html" (dict "kind" $kind "num" $num) }}{{ else if $anchor }}{{ warnf "shortcode %q: an anchor-only reference requires inner link text at %s; using the anchor as its text" .Name .Position }}{{ $label = $anchor }}{{ end -}} {{- $format := lower (.Page.Store.Get "tdOutputFormat" | default "html") -}} {{- if and (eq $format "print") (.Page.Store.Get "tdBookAggregate") $targetPage -}} {{- if $targetPage.Fragments.Identifiers.Contains $anchor -}} {{- $anchor = partial "book/aggregate-heading-anchor.html" (dict "page" $targetPage "anchor" $anchor) -}} {{- end -}} {{- $href = printf "#%s" $anchor -}} {{- end -}} {{- if eq $format "markdown" -}} {{- $markdownText := $inner -}} {{- if not $markdownText }}{{ $markdownText = $label | plainify }}{{ end -}} {{- if $missingTarget -}} {{- $markdownText | safeHTML -}} {{- else -}} {{- printf "[%s](%s)" $markdownText (partial "content/markdown-url.html" $href) | safeHTML -}} {{- end -}} {{- else if $missingTarget -}} {{ $label }} {{- else -}} {{ $label }} {{- end -}}
