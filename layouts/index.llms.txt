{{/* llms.txt indexes this site's own content. Menus routinely carry off-site
navigation (repository, issue tracker, chat); those are chrome, not content, and
listing them here dilutes the index and fails the markdown-link expectations
that agent-docs tooling applies to llms.txt entries. */ -}}
{{ $siteHost := (urls.Parse .Site.BaseURL).Host -}}
{{ range .Site.Menus.main -}}
  {{ $url := "" -}}
  {{ $description := "" -}}
  {{ $onSite := true -}}
  {{ with .Page -}}
    {{ $url = .Permalink -}}
    {{ with .OutputFormats.Get "markdown" -}}
      {{ $url = .Permalink -}}
    {{ end -}}
    {{ $description = .Description | strings.TrimSpace | replaceRE `\s+` " " -}}
  {{ else -}}
    {{ $parsedEntry := try (urls.Parse .URL) -}}
    {{ if $parsedEntry.Err -}}
      {{ $onSite = false -}}
    {{ else -}}
      {{ $u := $parsedEntry.Value -}}
      {{ if or (and $u.Scheme (not (in (slice "http" "https") $u.Scheme))) (hasPrefix .URL "#") -}}
        {{ $onSite = false -}}
      {{ else if and $u.Host (ne $u.Host $siteHost) -}}
        {{ $onSite = false -}}
      {{ else -}}
        {{ $url = cond (ne $u.Host "") .URL (strings.TrimPrefix "/" .URL | absLangURL) -}}
      {{ end -}}
    {{ end -}}
  {{ end -}}
  {{ if $onSite -}}
  - [{{ template "llms-label" .Name }}]({{ partial "content/markdown-url.html" $url }}){{ with $description }}: {{ . }}{{ end }} {{- "\n" -}}
  {{ end -}}
{{ end -}}

{{/* Documentation ----------------------------------------------------- */ -}}

{{/* The docs root is whatever params.ui.docs_section names, not a literal
"docs": shell/config.html is the single resolver for it. Every documentation
page is listed, not only the sections, because an index that stops at the
section level tells an agent nothing about what the site actually documents. */ -}}
{{ with .Site.GetPage (printf "/%s" $config.docsSection) -}}
  {{ $pages := .RegularPagesRecursive -}}
  {{ if or $pages .Sections -}}
    {{ "\n" -}}

    ## Documentation index

    {{- "\n\n" -}}
    {{ range .Sections -}}
      {{ template "llms-entry" (dict "page" . "depth" 0) -}}
      {{ range .RegularPagesRecursive -}}
        {{ template "llms-entry" (dict "page" . "depth" 1) -}}
      {{ end -}}
    {{ end -}}
    {{ range .RegularPages -}}
      {{ template "llms-entry" (dict "page" . "depth" 0) -}}
    {{ end -}}
  {{ end -}}
{{ end -}}

{{/* Full-text bundles -------------------------------------------------- */ -}}

{{/* Discovery belongs in the index file: list every top-level section that
opted into the LLMSFULL bundle in this language. Sites without bundles emit
nothing here. */ -}}
{{ $bundleLines := slice -}}
{{ range $section := .Site.Home.Sections -}}
  {{ with $section.OutputFormats.Get "llmsfull" -}}
    {{ $label := replace (replace (replace ($section.LinkTitle | default $section.Title | strings.TrimSpace | replaceRE `\s+` " ") "\\" "\\\\") "[" "\\[") "]" "\\]" -}}
    {{ $bundleLines = $bundleLines | append (printf "- [%s](%s)"
        $label (partial "content/markdown-url.html" .Permalink)) -}}
  {{ end -}}
{{ end -}}
{{ with $bundleLines -}}
  {{ "\n" -}}

  ## Full-text bundles

{{- define "llms-label" -}}
  {{- replace (replace (replace (printf "%v" . | strings.TrimSpace | replaceRE `\s+` " ") "\\" "\\\\") "[" "\\[") "]" "\\]" -}}
{{- end -}}

{{/* A page that Hugo never renders (`build.render: never`, sidebar dividers)
     has no permalink and must not become a dead link in the index. */ -}}
{{- define "llms-entry" -}}
  {{- $page := .page -}}
  {{- if not $page.Params.sidebar_divider -}}
  {{- with $page.Permalink -}}
    {{- $url := . -}}
    {{- with $page.OutputFormats.Get "markdown" -}}
      {{- $url = .Permalink -}}
    {{- end -}}
    {{- $title := replace (replace (replace ($page.LinkTitle | default $page.Title | strings.TrimSpace | replaceRE `\s+` " ") "\\" "\\\\") "[" "\\[") "]" "\\]" -}}
    {{- $description := $page.Description | strings.TrimSpace | replaceRE `\s+` " " -}}
    {{- $indent := strings.Repeat $.depth "  " -}}
    {{- printf "%s- [%s](%s)" $indent $title (partial "content/markdown-url.html" $url) -}}
    {{- with $description }}: {{ . }}{{ end -}}
    {{- "\n" -}}
  {{- end -}}
  {{- end -}}
{{- end -}}
