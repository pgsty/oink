{{/* llms.txt indexes this site's own content. Menus routinely carry off-site
navigation (repository, issue tracker, chat); those are chrome, not content, and
listing them here dilutes the index and fails the markdown-link expectations
that agent-docs tooling applies to llms.txt entries. */ -}}
{{ $siteHost := (urls.Parse .Site.BaseURL).Host -}}
{{ range .Site.Menus.main -}}
  {{ $url := "" -}}
  {{ $description := "" -}}
  {{ $onSite := true -}}
  {{ with .Page -}}
    {{ $url = .Permalink -}}
    {{ with .OutputFormats.Get "markdown" -}}
      {{ $url = .Permalink -}}
    {{ end -}}
    {{ $description = .Description | strings.TrimSpace | replaceRE `\s+` " " -}}
  {{ else -}}
    {{ $entryHost := (urls.Parse .URL).Host -}}
    {{ if and $entryHost (ne $entryHost $siteHost) -}}
      {{ $onSite = false -}}
    {{ else -}}
      {{ $url = cond (ne $entryHost "") .URL (.URL | absLangURL) -}}
    {{ end -}}
  {{ end -}}
  {{ if $onSite -}}
  - [{{ .Name }}]({{ $url }}){{ with $description }}: {{ . }}{{ end }} {{- "\n" -}}
  {{ end -}}
{{ end -}}

{{/* Documentation ----------------------------------------------------- */ -}}

{{/* The docs root is whatever params.ui.docs_section names, not a literal
"docs": shell/config.html is the single resolver for it. Every documentation
page is listed, not only the sections, because an index that stops at the
section level tells an agent nothing about what the site actually documents. */ -}}
{{ with .Site.GetPage (printf "/%s" $config.docsSection) -}}
  {{ $pages := .RegularPagesRecursive -}}
  {{ if or $pages .Sections -}}
    {{ "\n" -}}

    ## Documentation index

    {{- "\n\n" -}}
    {{ range .Sections -}}
      {{ template "llms-entry" (dict "page" . "depth" 0) -}}
      {{ range .RegularPagesRecursive -}}
        {{ template "llms-entry" (dict "page" . "depth" 1) -}}
      {{ end -}}
    {{ end -}}
    {{ range .RegularPages -}}
      {{ template "llms-entry" (dict "page" . "depth" 0) -}}
    {{ end -}}
  {{ end -}}
{{ end -}}

{{/* Multilingual sites ------------------------------------------------------- */ -}}

{{ with hugo.Sites -}}
  {{ "\n" -}}

  ## Site locales

  {{- "\n\n" -}}
  {{ range . -}}
    {{ $home := .Home -}}
    {{ $url := $home.Permalink -}}
    {{ with $home.OutputFormats.Get "markdown" -}}
      {{ $url = .Permalink -}}
    {{ end -}}
    - [{{ $home.Language.Label }}]({{ $url }}) {{- "\n" -}}
  {{ end -}}
{{ end -}}

{{/* A page that Hugo never renders (`build.render: never`, sidebar dividers)
     has no permalink and must not become a dead link in the index. */ -}}
{{- define "llms-entry" -}}
  {{- $page := .page -}}
  {{- with $page.Permalink -}}
    {{- $url := . -}}
    {{- with $page.OutputFormats.Get "markdown" -}}
      {{- $url = .Permalink -}}
    {{- end -}}
    {{- $title := $page.LinkTitle | default $page.Title | strings.TrimSpace -}}
    {{- $description := $page.Description | strings.TrimSpace | replaceRE `\s+` " " -}}
    {{- $indent := strings.Repeat $.depth "  " -}}
    {{- printf "%s- [%s](%s)" $indent $title $url -}}
    {{- with $description }}: {{ . }}{{ end -}}
    {{- "\n" -}}
  {{- end -}}
{{- end -}}
